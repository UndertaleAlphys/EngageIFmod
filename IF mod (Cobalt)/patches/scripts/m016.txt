function InitDragonVein()
    EffectCreate("魔法陣_G001", 26, 11)
    DraginVein_Skill_Equip( "SID_DragonVein_M016_TIDE" )
end

-- -----------------------------------
function DragonVeinUsed()
    local unit = MindGetUnit();
    local x = UnitGetX(unit)
    local z = UnitGetZ(unit)
    local used = ""
    local DragonVeinType = ""
    if(x == 26 and z == 11) then
        used = DraginVein_Skill_Remove("SID_DragonVein_G004_ICE_0","SID_DragonVein_Used")
        DragonVeinType = "tide"
    end

    if used == true then
        WaitTime( 0.5 )
        EffectPlay( "カムイ竜脈", x, z )
        WaitTime( 3.0 )
        EffectDelete("魔法陣_G001", x, z)
        TerrainSetBegin()
            TerrainSet( x, z, "TID_平地")
        TerrainSetEnd()
    end
    Log("DragonVeinUsedComplete")
end

function BattleStartAfterイベント()

	-- 緑軍→青軍
	UnitJoin( "PID_ロサード", "PID_ゴルドマリー" )
	WaitTime( 0.5 )

	CursorSetPos_FromPid( "PID_ゴルドマリー" )

	Talk( "MID_EV2" )

	local x = UnitGetX(g_pid_Marron)
    local z = UnitGetZ(g_pid_Marron)
    CursorSetPos_FromPid( g_pid_Marron )
    WaitTime( 0.5 )
    EffectPlay( "水球_発動", x, z )
    EffectWait()
    HighTide()

end

function condition_HighTide()

	local turn = VariableGet( g_key_highTide_turn )

	if turn == 1 then
		VariableSet( g_key_highTide_turn, 0 )
		return true
	else
		VariableSet( g_key_highTide_turn, turn + 1 )
		return false
	end

end

function HighTide()
    MapObjectAction( 9, 1, MAP_ACTION_IDLE)
    マップ全体の地形変化( "TID_浅瀬", "TID_砂浜_M016" )

    VariableSet( g_key_highTide, 0 )
end

function マップ全体の地形変化( fromTid, toTid )

	TerrainSetBegin()
	for x = 0, map_width-1 do
		for z = 0, map_height-1 do
			if TerrainGet(x, z) == fromTid then
				TerrainSet(x, z, toTid)
			end
		end
	end
	TerrainSetEnd()

end

function condition_蛮族登場()

	if VariableGet( g_key_bandit ) != 0 then
		return false
	end

	if DifficultyGet() > DIFFICULTY_NORMAL then
		return true
	end

	return false

end

function condition_蛮族登場_ノーマル()

	if VariableGet( g_key_bandit ) != 0 then
		return false
	end

	if DifficultyGet() == DIFFICULTY_NORMAL then
		return true
	end

	return false

end

function 蛮族登場()

	Dispos( "Bandit", DISPOS_FLAG_FOCUS )
	Yield()
	WaitTime( 0.5 )

	VariableSet( g_key_bandit, 1 )

end
