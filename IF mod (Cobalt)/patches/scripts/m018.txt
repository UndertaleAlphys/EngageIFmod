function Startup()
	
	Log("Startup");
	
	-- 勝利条件設定：敵将撃破
	WinRuleSetEnemyNumberLessThanOrEqualTo(-1)
	WinRuleSetMID( "MID_RULE_M018_WIN" )

	変数登録()
	イベント登録();
	

end

function 変数登録()

	VariableEntry("TalkToThiefTimes", 0)
	VariableEntry("DefeatThiefTimes", 0)
	VariableEntry("VisitedTopLeft", 0)
	VariableEntry("VisitedTopRight", 0)
	VariableEntry("VisitedBottomLeft", 0)
	VariableEntry("VisitedBottomRight", 0)

end

	EventEntryDie(ProcessDeadCount, "PID_M018_イルシオン兵_ボス", FORCE_ENEMY, condition_true);

	EventEntryDie(EmptyFunction, "PID_リンデン", FORCE_ALL, "S_リンデン_死亡_済")

    EventEntryTalk(TalkToThiefM018,"",FORCE_PLAYER, "PID_IFMOD_M018_イルシオン兵_上忍_Ally3", FORCE_ALLY,Condition_CanTalkBottomLeft)

function TalkToThiefM018()
    local TalkTimes = VariableGet("TalkToThiefTimes")
    TalkTimes = TalkTimes + 1
    VariableSet("TalkToThiefTimes",TalkTimes)
    local unit = MindGetTargetUnit()
    local x = UnitGetX(unit)
    local z = UnitGetZ(unit)
    EffectPlay( "竜化_G1" , x, z )
    WaitTime( 0.5 )
    UnitDelete( unit )
    if TalkTimes ~= 4 then
        if x == 6 and z == 24 then
            Dispos( "NormalBossTopLeft", DISPOS_FLAG_FOCUS + DISPOS_FLAG_FORCED )
            VariableSet("VisitedTopLeft",1)
        end
        if x == 24 and z == 26 then
            Dispos( "NormalBossTopRight", DISPOS_FLAG_FOCUS + DISPOS_FLAG_FORCED )
            VariableSet("VisitedTopRight",1)
        end
        if x == 6 and z == 3 then
            Dispos( "NormalBossBottomLeft", DISPOS_FLAG_FOCUS + DISPOS_FLAG_FORCED )
            VariableSet("VisitedBottomLeft",1)
        end
        if x == 24 and z == 6 then
            Dispos( "NormalBossBottomRight", DISPOS_FLAG_FOCUS + DISPOS_FLAG_FORCED )
            VariableSet("VisitedBottomRight",1)
        end
    else
        if x == 6 and z == 24 then
            Dispos( "EnhanceBossTopLeft", DISPOS_FLAG_FOCUS + DISPOS_FLAG_FORCED )
            VariableSet("VisitedTopLeft",1)
        end
        if x == 24 and z == 26 then
            Dispos( "EnhanceBossTopRight", DISPOS_FLAG_FOCUS + DISPOS_FLAG_FORCED )
            VariableSet("VisitedTopRight",1)
        end
        if x == 6 and z == 3 then
            Dispos( "EnhanceBossBottomLeft", DISPOS_FLAG_FOCUS + DISPOS_FLAG_FORCED )
            VariableSet("VisitedBottomLeft",1)
        end
        if x == 24 and z == 6 then
            Dispos( "EnhanceBossBottomRight", DISPOS_FLAG_FOCUS + DISPOS_FLAG_FORCED )
            VariableSet("VisitedBottomRight",1)
        end
    end
end

function Condition_CanTalkTopLeft()
    return VariableGet("VisitedTopLeft") == 0
end

function Condition_CanTalkTopRight()
    return VariableGet("VisitedTopRight") == 0
end

function Condition_CanTalkBottomLeft()
    return VariableGet("VisitedBottomLeft") == 0
end

function Condition_CanTalkBottomRight()
    return VariableGet("VisitedBottomRight") == 0
end

function ProcessDeadCount()
    local DeadTimes = VariableGet("DefeatThiefTimes")
    DeadTimes = DeadTimes + 1;
    VariableSet("DefeatThiefTimes",DeadTimes)
    if DeadTimes == 4 then
       VariableSet( "勝利", 1 )
    end
end

function StealMoney()
    local StealMoneyValue = 500 * (4-VariableGet("DefeatThiefTimes"))
    local NowMoney = GoldGet()
    NowMoney = NowMoney - StealMoneyValue
    MessSetArgument(0,StealMoneyValue)
    MessSetArgument(1,NowMoney)
    Dialog("MID_TUT_M018_TURNPROCESS")
    GoldSet(NowMoney)
end

function リンデン加入()
	pid = "PID_リンデン";
	if UnitExistOnMap( pid ) then
		UnitJoin( pid );
		
		-- リンデンから「SID_死亡会話存在敵」を削除
		スキル解除( "PID_リンデン", "SID_死亡会話存在敵" )
	end
end

function 増援６ターン目()
	
	if DifficultyGet() == DIFFICULTY_NORMAL then
		return false
	end
	
	Dispos("Enemy_Reinforcement0", DISPOS_FLAG_FOCUS);
	Yield();
	WaitTime(0.5);
end

-- -----------------------------------

function 離脱イベント()
	local unit = MindGetUnit()
	if not UnitExistOnMap( unit ) then
		return
	end
	
	UnitRotation( g_pid_lueur, ROTATE_UP)
	UnitMoveWait()
	
	VariableSet( "勝利", 1 )
end
