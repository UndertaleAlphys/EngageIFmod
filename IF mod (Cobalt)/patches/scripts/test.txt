function CamillaAiSetTurnBefore()
    -- 检查卡米拉AI是否已经完全激活
    if VariableGet(g_key_camillaActive) == 1 then
        Log("Already Active")
        return
    end

    local attackRangeCamilla = GetUnitActualAttackRange(g_pid_camilla)

    local index = ForceUnitGetFirst(FORCE_PLAYER)

    while index ~= nil do
        local indexInRange = IsUnitInAttackRange(attackRangeCamilla,index)
        if indexInRange == true then
            Log("Active")
            AiSetSequence(g_pid_camilla, AI_ORDER_CAUSE, "AI_AC_Everytime")
            VariableSet(g_key_camillaActive,  1)
            return
        end
        index = ForceUnitGetNext(index)
    end

    AiSetSequence(g_pid_camilla, AI_ORDER_CAUSE, "AI_AC_Null")
    Log("Inactive")
end
