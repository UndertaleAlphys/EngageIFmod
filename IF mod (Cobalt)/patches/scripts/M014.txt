function 召喚渦作成(areaname)
	CursorSetPos_FromPid(g_pid_boss)
	MapCameraWait()

	local countD = DisposGetGroupCount(areaname)
	countD = countD - 1

		-- グループの人数分処理を回し、ユニットの座標を走査、取得する
		for i = 0, countD do
			local unitX = DisposGetUnitX(areaname, i,DifficultyGet())
			local unitZ = DisposGetUnitZ(areaname, i,DifficultyGet())

			if unitX != -1 then
				EffectCreate( "召喚渦", unitX,unitZ )
			end
		end

end

function 召喚渦消去と配置(areaname)
	CursorSetPos_FromPid(g_pid_boss)
	MapCameraWait()

	Dispos( areaname, DISPOS_FLAG_FOCUS + DISPOS_FLAG_WARP  + DISPOS_FLAG_NOT_FORCED)
	Yield()

	countD = DisposGetGroupCount(areaname)
	countD = countD - 1
	for i = 0, countD do
		local unitX = DisposGetUnitX(areaname, i,DifficultyGet())
		local unitZ = DisposGetUnitZ(areaname, i,DifficultyGet())
		if unitX != -1 then
			EffectDelete( "召喚渦", unitX,unitZ )
		end
	end
	Yield()

end

function DragonVeinEffect()
	local index	= ForceUnitGetFirst( FORCE_ENEMY )
	while ( index != nil ) do
		local pid = UnitGetPID( index )
		UnitSetPrivateSkill(index, "SID_移動不可")
		index = ForceUnitGetNext( index )		
	end
end

    SetAIForAllEnemies()
end

function SetAIForAllEnemies()
    --Log("Setting AI for all enemy units")

    local count = 0
    local unit = ForceUnitGetFirst(FORCE_ENEMY)

    while unit ~= nil do
        local pid = UnitGetPID(unit)

        AiSetSequence(unit, AI_ORDER_CAUSE, "AI_AC_Everytime")
        AiSetSequence(unit, AI_ORDER_MOVE, "AI_MV_NearestEnemy")
        AiSetActive(unit, true)

        --Log("Set AI for enemy unit: " .. pid)

        count = count + 1
        unit = ForceUnitGetNext(unit)
    end

    --Log("Finished setting AI for " .. count .. " enemy units")
end

	local unit = MindGetUnit()

	end
	
	UnitRotation( g_pid_lueur, ROTATE_UP)

end
