function InitIvy()
    MapOverlapSetBegin()
    for z = 1, g_Height do
        for x = 1, g_Width do
            local key = x + (z) * g_Width
            if g_TerraSheetMap[key] != " " then
                local reverseZ = g_Height + 1 - z
                MapOverlapSet(x, reverseZ, "TID_ツタ_G002")
            end
        end
    end
    MapOverlapSetEnd()
end

function UpdateIvyMapRange()
    MapRangeClear()

    local turn = MapGetTurn()
    local nextTurn = turn + 1

    MapRangeAddBegin()
    for z = 1, g_Height do
        for x = 1, g_Width do
            local key = x + z * g_Width
            local reverseZ = g_Height + 1 - z

            local currentHasIvy = false
            if(turn % 3 == 1) then
                if g_TerraSheetMap[key] == "2" or g_TerraSheetMap[key] == "3" then
                    currentHasIvy = true
                end
            elseif(turn % 3 == 2) then
                if g_TerraSheetMap[key] == "1" or g_TerraSheetMap[key] == "3" then
                    currentHasIvy = true
                end
            else
                if g_TerraSheetMap[key] == "1" or g_TerraSheetMap[key] == "2" then
                    currentHasIvy = true
                end
            end

            local nextHasIvy = false
            if(nextTurn % 3 == 1) then
                if g_TerraSheetMap[key] == "2" or g_TerraSheetMap[key] == "3" then
                    nextHasIvy = true
                end
            elseif(nextTurn % 3 == 2) then
                if g_TerraSheetMap[key] == "1" or g_TerraSheetMap[key] == "3" then
                    nextHasIvy = true
                end
            else
                if g_TerraSheetMap[key] == "1" or g_TerraSheetMap[key] == "2" then
                    nextHasIvy = true
                end
            end

            if not currentHasIvy and nextHasIvy then
                MapRangeAdd(x, reverseZ)
            end
        end
    end
    MapRangeAddEnd()
 end

function SetLynnAIMoveToFarthestPlayer()
    local lynnPid = "PID_G002_IFMOD_リン"

    if not UnitIsExist(lynnPid) then
        return
    end

    local farthestPlayer = nil
    local maxDistance = -1
    local lynnX = UnitGetX(lynnPid)
    local lynnZ = UnitGetZ(lynnPid)

    local playerUnit = ForceUnitGetFirst(FORCE_PLAYER)
    while playerUnit ~= nil do
        local playerPid = UnitGetPID(playerUnit)

        if playerPid ~= lynnPid then
            local playerX = UnitGetX(playerUnit)
            local playerZ = UnitGetZ(playerUnit)

            local distance = math.abs(lynnX - playerX) + math.abs(lynnZ - playerZ)

            if distance > maxDistance then
                maxDistance = distance
                farthestPlayer = playerPid
            end
        end

        playerUnit = ForceUnitGetNext(playerUnit)
    end

    if farthestPlayer ~= nil then
        AiSetSequence(lynnPid, AI_ORDER_MOVE, "AI_MV_Person", farthestPlayer)
    end
end

	WaitTime( 1.5 )

	-- ヴァンドレによる毒ガスへの警戒会話を流す
	Talk("MID_EV2")

	CursorAnimeCreate( 21, 4, "W2H3" )
	CursorAnimeDelete()

	WaitTime( 0.5 )

	Talk("MID_EV5")
	
	WaitTime( 0.5 )
	
	SetIvy( 1 )

	WaitTime( 1.5 )

	SetIvy( 2 )

	WaitTime( 1.5 )

	SetIvy( 3 )

	WaitTime( 1.5 )

	Talk("MID_EV6")
end

function condition_リンと戦闘()

	-- 既に再生済みの場合は二度と再生しない
	if VariableGet( "戦闘前会話_ヘクトル_リン_済" ) == 1 then
		return false
	end
	
	-- 戦闘に参加した味方がつけている紋章士のGIDを取得する
	local god = nil
	if MindGetForce() == FORCE_PLAYER then
		god = UnitGetGodUnit( MindGetUnit() )
	else
		god = UnitGetGodUnit( MindGetTargetUnit() )
	end

	if ( god == "GID_リン" ) then
		return true
	else
		return false
	end
	
end
