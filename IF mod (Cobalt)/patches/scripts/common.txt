-- 一定時間停止(1秒=1.0)
function WaitTime(time)
	while time > 0 do
		if SkipIsBlackOut() then
			break	--スキップ
		end
		-- 時間経過
		time = time - TimeGetDelta()
		-- 待ち
		coroutine.yield(true)
	end
end

-- キー入力待ち(デバッグ用)
function WaitKeyForDebug()
	while true do
		if SkipIsBlackOut() then
			break	--スキップ
		end
		if DebugIsButton(DEBUG_BUTTON_A) then
			break	--キー入力
		end
		if DebugIsButton(DEBUG_BUTTON_Y) and DebugIsTrigger(DEBUG_BUTTON_X) then
			DebugCreateMenu()
		end
		coroutine.yield(true)
	end
end

-- 最小値取得
function Min(value, min)
	if value > min then
		return min
	else
		return value
	end
end

-- ホワイトアウト
function WhiteOut(time)
	FadeOut(time, 1, 1, 1)
end

-- 最大値取得
function Max(value, max)
	if value < max then
		return max
	else
		return value
	end
end

-- クランプ
function Clamp(value, min, max)
	value = Min(value, max)
	value = Max(value, min)
	return value
end

-- 変数の加算
function VariableAdd(key, value, min, max)
	value = VariableGet(key) + value
	value = Clamp(value, min, max)
	VariableSet(key, value)
	return value
end

-- 変数の加算(+1)
function VariableInc(key, min, max)
	return VariableAdd(key, 1, min, max)
end

-- 変数の減算(-1)
function VariableDec(key, min, max)
	return VariableAdd(key, -1, min, max)
end

--軍のユニット数を取得
function ForceUnitGetCount(force)
	local count = 0
	local index = ForceUnitGetFirst(force)
	while index != nil do
		index = ForceUnitGetNext(index)	
		count = count + 1
	end
	return count
end

-- ターゲットへの射程判定
function 射程判定(unit, target)

	if UnitIsExist(unit) and UnitIsExist(target) then

		local sx = UnitGetX(unit)
		local sz = UnitGetZ(unit)

		local tx = UnitGetX(target)
		local tz = UnitGetZ(target)

		--相手のとの距離を取得
		local distance = math.abs(tx - sx) + math.abs(tz - sz)
		
		--武器を取得
		for i = 0 , UnitGetItemCount(unit) do
			--武器の射程を取得
			local rangeI = UnitGetItemRangeI(unit, i)
			local rangeO = UnitGetItemRangeO(unit, i)
			--射程判定
			if rangeI <= distance and distance <= rangeO then
				return true
			end
		end
		
	end

	return false
end

--Callコマンド追加
function MenuAddCall(menu, mid, func, condition, args)
	local item = MenuItemCreate(menu)
	MenuItemSetMid(item, mid)
	MenuItemSetFunc(item, func, args)
	MenuItemSetCondition(item, condition)
	return item
end

--Jumpコマンド追加
function MenuAddJump(menu, mid, func, condition)
	return MenuAddCall(menu, mid, Jump, condition, func)
end

--Talkコマンド追加
function MenuAddTalk(menu, mid, talkId, condition)
	local item = MenuAddCall(menu, mid, Talk, condition, talkId)
	return item
end

--Backコマンド追加
function MenuAddBack(menu, mid, func, condition)
	MenuCancelJump(menu, func, condtion)
	return MenuAddJump(menu, mid, func, condtion)
end

--選択コマンド追加
function MenuAddCallWithObjectFlash(menu, mid, func, condition, objName)
	local item = MenuAddCall(menu, mid, func, condition)
	MenuItemSetSelectFunc(item, ObjectFlash, objName)
	return item
end

--メニューアイテムの有無をチェック
function MenuItemExist(mid)
	--ラベルが無い
	if MessIsExist(mid) == false then
			return false;
	end
	--メニューラベル名で定義済み変数が非表示なら隠す
	if VariableIsExist(mid) then
		if VariableGet(mid) == MENU_ITEM_HIDE then
			return false;
		end
	end
	return true;
end

--メニューラベル追加
function TryMenuAddLabel(menu, label)
	if MenuItemExist(label) then
		MenuAddLabel(menu, label)
	end
end

--会話コマンド追加
function TryMenuAddTalk(menu, selectId, talkId)
	if MenuItemExist(selectId) then
		MenuAddTalk(menu, selectId, talkId, condition)
	end
end

function CursorAnimeCreate( x, z, ... )
	
	local size = "W1H1"
	local name = "Eff_Cursor01"
	
	local var = {...}
	if #var >= 1 then
		size = var[1]
	end
	if #var >= 2 then
		name = var[2]
	end
	
	-- --------------------
	
	if ( size == "W1H1" ) then
		CursorSetPos(x,z)
		MapCameraWait()
	end
	MapObjectCreate(name, "Effects/BMap/UI/Guide/Prefabs/Eff_Cursor_" .. size, x, z)
	WaitTime( 2.0 )
	
end

--プレイヤー軍の中心点を算出します
----------------------------------------------
function プレイヤー軍の中心点を算出()
	local x = 0
	local z = 0
	local count = 0
	
	local index = ForceUnitGetFirst(FORCE_PLAYER)
	while index != nil do
		
		x = ( x + UnitGetX(index) )
		z = ( z + UnitGetZ(index) )
		
		count = count + 1
		index = ForceUnitGetNext(index)
	end
	
	local center_x = math.floor(x/count + 0.5);
	local center_z = math.floor(z/count + 0.5);
	
	return center_x, center_z
end

function 加勢ポイントのエリアイベント登録( list )

		local flag = "加勢ポイント_" .. tostring( x ) .. "_" .. tostring( z ) .. "_エフェクト展開_済み"
		
		local option	= HELP_POINT_ALL
		if #list[pointer] >= 3 then

		end
		
		EventEntryArea(加勢ポイントのエフェクト展開, x, z, x, z, FORCE_PLAYER, flag, x, z, option )
	end

function 加勢ポイントのエフェクト展開( x, z, option )

	-- この処理に数フレームかかるのでスキップを無効にする
	-- (必ず SkipPopState() と対にすること)
	SkipPushStateAndDisable()
	
	-- ===================================================================

	EventStateObject( x, z, 1 )
	
	-- ===================================================================
	
	local isLeft = ( option / HELP_POINT_LEFT >= 1 )
	option = option % HELP_POINT_LEFT
	
	local isDown = ( option / HELP_POINT_DOWN >= 1 )
	option = option % HELP_POINT_DOWN
	
	local isRight = ( option / HELP_POINT_RIGHT >= 1 )
	option = option % HELP_POINT_RIGHT
	
	local isUp = ( option / HELP_POINT_UP >= 1 )
	option = option % HELP_POINT_UP
	
	-- ===================================================================
	
	if isUp then
		EffectPlay( "加勢ポイント_発動", x, z+1 )
	end
	
	if isRight then
		EffectPlay( "加勢ポイント_発動", x+1, z )
	end
	
	if isDown then
		EffectPlay( "加勢ポイント_発動", x, z-1 )
	end
	
	if isLeft then
		EffectPlay( "加勢ポイント_発動", x-1, z )
	end
	
	-- ===================================================================
	
	WaitTime( 0.2 )
	
	-- ===================================================================
	
	if isUp then
		EffectCreate( "加勢ポイント_ループ", x, z+1 )
	end
	
	if isRight then
		EffectCreate( "加勢ポイント_ループ", x+1, z )
	end
	
	if isDown then
		EffectCreate( "加勢ポイント_ループ", x, z-1 )
	end
	
	if isLeft then
		EffectCreate( "加勢ポイント_ループ", x-1, z )
	end
	
	-- ===================================================================
	
	-- スキップを無効にしていたのを元に戻す
	-- (必ず SkipPushStateAndDisable() と対にすること)
	SkipPopState()

end

function getRangeSquare(squareSize,map_height,map_width)

    local results = {}

    for z = 1, map_height - squareSize + 1 do
        for x = 1, map_width - squareSize + 1 do
            local playerUnitCount = 0

            for sz = z, z + squareSize - 1 do
                for sx = x, x + squareSize - 1 do
                    local unit = UnitGetByPos(sx, sz)
                    if unit != nil and UnitGetForce(unit) == FORCE_PLAYER then
                        playerUnitCount = playerUnitCount + 1
                    end
                end
            end

            local centerX = x + (squareSize - 1) * 0.5
            local centerZ = z + (squareSize - 1) * 0.5
            results[#results + 1] = {
                x = x,
                z = z,
                centerX = centerX,
                centerZ = centerZ,
                playerUnitCount = playerUnitCount
            }

        end
    end

    return results
end

function GetUnitReachablePositions(unit)
    local unitX = UnitGetX(unit)
    local unitZ = UnitGetZ(unit)
    local moveType = UnitGetMoveCost(unit)
    local moveValue = UnitGetCapability(unit, CAPABILITY_MOVE, true)

    local reachablePositions = {}
    local visited = {}

    local queue = {{x = unitX, z = unitZ, remainingMove = moveValue}}

    -- Init first position
    table.insert(reachablePositions, {x = unitX, z = unitZ})

    visited[unitX .. "," .. unitZ] = true
    -- do bfs
    while #queue > 0 do
        local current = table.remove(queue, 1)

        local directions = {{1, 0}, {-1, 0}, {0, 1}, {0, -1}}
        for _, dir in ipairs(directions) do

            local newX = current.x + dir[1]
            local newZ = current.z + dir[2]

            -- position in map range
            if newX > 0 and newX < g_map_width and newZ > 0 and newZ < g_map_height then
                local posKey = newX .. "," .. newZ

                -- have not been visited
                if not visited[posKey] then
                    local terrainCost = GetTerrainMoveCost(newX, newZ, moveType)

                    local unitAtPos = UnitGetByPos(newX, newZ)
                    local canPass = CanPassThrough(unit, unitAtPos)
                    Log(current.remainingMove)
                    if current.remainingMove >= terrainCost and canPass then

                        local newRemainingMove = current.remainingMove - terrainCost

                        table.insert(queue, {x = newX, z = newZ, remainingMove = newRemainingMove})

                        table.insert(reachablePositions, {x = newX, z = newZ})

                        visited[posKey] = true

                    end

                end
            end
        end
    end

    -- Finally remove position which forcePlayer stand at
    local filteredPositions = {}
    for _, pos in ipairs(reachablePositions) do
        local unitAtPos = UnitGetByPos(pos.x, pos.z)
        if unitAtPos == nil or UnitGetForce(unitAtPos) != FORCE_PLAYER then
            table.insert(filteredPositions, pos)
        end
    end

    return filteredPositions

end

function GetTerrainMoveCost(x, z, moveType)
    local terrainCost = TerrainGetMoveCost(x, z)

    if terrainCost == "COST_不可" then
        return 255
    elseif terrainCost == "COST_空" then
        if moveType == "COST_飛行" then
            return 1
        else
            return 255
        end
    elseif terrainCost == "COST_平地" then
        return 1
    elseif terrainCost == "COST_林" then
        return 2
    elseif terrainCost == "COST_浅瀬" then
        return 3
    elseif terrainCost == "COST_竜のみ" then
        if moveType == "COST_竜" then
            return 1
        else
            return 255
        end
    else
        return 1
    end
end

function CanPassThrough(movingUnit, unitAtPos)
    if unitAtPos == nil then
        return true
    end

    local movingUnitForce = UnitGetForce(movingUnit)
    local targetUnitForce = UnitGetForce(unitAtPos)

    if movingUnitForce == FORCE_ENEMY and targetUnitForce == FORCE_ENEMY then
        return true
    end

    if movingUnitForce == FORCE_PLAYER and targetUnitForce == FORCE_ENEMY then
        if UnitHasWholeSkill(movingUnit, "SID_すり抜け") then
            return true
        end
        return false
    end

    if movingUnitForce == FORCE_ENEMY and targetUnitForce == FORCE_PLAYER then
        if UnitHasWholeSkill(movingUnit, "SID_すり抜け") then
            return true
        end
        return false
    end

    return true
end

function GetUnitActualAttackRange(unit)
    local reachablePositions = GetUnitReachablePositions(unit)

    local weaponCount = UnitGetItemCount(unit)

    local attackRange = {}
    local attackRangeMap = {} -- 用于去重

    for _, pos in ipairs(reachablePositions) do
        local posX = pos.x
        local posZ = pos.z

        for i = 0, weaponCount - 1 do
            local rangeMin = UnitGetItemRangeI(unit, i)
            local rangeMax = UnitGetItemRangeO(unit, i)

            for dx = -rangeMax, rangeMax do
                for dz = -rangeMax, rangeMax do
                    local distance = math.abs(dx) + math.abs(dz)
                    if distance >= rangeMin and distance <= rangeMax then
                        local targetX = posX + dx
                        local targetZ = posZ + dz

                        if targetX > 0 and targetX < g_map_width and
                           targetZ > 0 and targetZ < g_map_height then

                            local posKey = targetX .. "," .. targetZ
                            if not attackRangeMap[posKey] then
                                attackRangeMap[posKey] = true
                                table.insert(attackRange, {x = targetX, z = targetZ})
                            end
                        end
                    end
                end
            end
        end
    end

    return attackRange
end

function IsUnitInAttackRange(attackRange, targetUnit)
    local targetX = UnitGetX(targetUnit)
    local targetZ = UnitGetZ(targetUnit)

    for _, pos in ipairs(attackRange) do
        if pos.x == targetX and pos.z == targetZ then
            return true
        end
    end

    return false
end

function getUnitsUnderEnemyAttackCount()
    local unitAttackCount = {}

    local playerUnit = ForceUnitGetFirst(FORCE_PLAYER)
    while playerUnit != nil do
        local pid = UnitGetPID(playerUnit)
        unitAttackCount[pid] = {
            pid = pid,
            count = 0
        }
        playerUnit = ForceUnitGetNext(playerUnit)
    end

    local enemyUnit = ForceUnitGetFirst(FORCE_ENEMY)
    while enemyUnit != nil do
        local enemyAttackRange = GetUnitActualAttackRange(enemyUnit)

        for _, pos in ipairs(enemyAttackRange) do
            local unitAtPos = UnitGetByPos(pos.x, pos.z)
            if unitAtPos != nil and UnitGetForce(unitAtPos) == FORCE_PLAYER then
                local pid = UnitGetPID(unitAtPos)
                if unitAttackCount[pid] then
                    unitAttackCount[pid].count = unitAttackCount[pid].count + 1
                end
            end
        end

        enemyUnit = ForceUnitGetNext(enemyUnit)
    end

    return unitAttackCount
end

function ShouldUnitLand(unit)
    local x = UnitGetX(unit)
    local z = UnitGetZ(unit)
    local terrainCost = TerrainGetMoveCost(x, z)

    -- 当地形为不可通行时，无论什么移动类型都需要降落
    if terrainCost == "COST_不可" then
        return true
    end

    -- 当地形为空中时，非飞行单位需要降落
    if terrainCost == "COST_空" then
        local moveType = UnitGetMoveCost(unit)
        if moveType ~= "COST_飛行" then
             return true
        end
    end

    return false
end


function FindLandingPosition(startX, startZ,isFly)
    local queue = {{x = startX, z = startZ}}
    local visited = {}
    local key = startX + startZ * g_width
    visited[key] = true

    -- 四个方向：上、右、下、左
    local directions = {{0, -1}, {1, 0}, {0, 1}, {-1, 0}}

    -- BFS搜索
    while #queue > 0 do
        local current = table.remove(queue, 1)
        local currentX, currentZ = current.x, current.z

        -- 检查当前位置是否适合降落
        if IsSuitableLandingPosition(currentX, currentZ, isFly) then
            return currentX, currentZ
        end

        -- 检查四个方向的相邻点
        for _, dir in ipairs(directions) do
            local nextX = currentX + dir[1]
            local nextZ = currentZ + dir[2]

            -- 检查边界
            if nextX >= 0 and nextX < g_width and nextZ >= 0 and nextZ < g_height then
                local nextKey = nextX + nextZ * g_width

                -- 如果未访问过，则加入队列
                if not visited[nextKey] then
                    table.insert(queue, {x = nextX, z = nextZ})
                    visited[nextKey] = true
                end
            end
        end
    end

    -- 如果没有找到合适位置，返回原始位置
    return startX, startZ
end

function IsSuitableLandingPosition(x, z ,isFly)
    -- 检查地形是否不为COST_不可和COST_空
    local terrainCost = TerrainGetMoveCost(x, z)
    if isFly == false then
       if terrainCost == "COST_不可" or terrainCost == "COST_空" then
           return false
       end
    end

    if isFly == true then
        if terrainCost == "COST_不可" then
           return false
       end
    end

    -- 检查位置是否没有其他人
    if UnitGetByPos(x, z) == nil then
        return true
    end

    return false
end

function LandUnit(unit)
    if ShouldUnitLand(unit) then
        local x = UnitGetX(unit)
        local z = UnitGetZ(unit)
        local isFlyBattleType = (UnitGetMoveCost(unit) == "COST_飛行")
        -- 查找合适的降落位置
        local landX, landZ = FindLandingPosition(x, z ,isFlyBattleType)

        -- 如果找到不同的位置，则移动单位
        if landX ~= x or landZ ~= z then
             -- 保存单位移动前的位置历史记录
            MapHistoryPositionListBegin()
            MapHistoryPositionList(unit)
            MapHistoryPositionListEnd()

            CursorSetPos(landX,landZ)
            MapCameraWait()
            -- 使用跳躍方式移动单位
            UnitJumpPos(unit, landX, landZ, 0.5)
            UnitMoveWait()
            return true
        end
    end

    return false
end

function ProcessAllUnitsLanding()
    -- 检查我方单位
    local playerIndex = ForceUnitGetFirst(FORCE_PLAYER)
    while playerIndex ~= nil do
        LandUnit(playerIndex)
        playerIndex = ForceUnitGetNext(playerIndex)
    end

    -- 检查敌方单位
    local enemyIndex = ForceUnitGetFirst(FORCE_ENEMY)
    while enemyIndex ~= nil do
        LandUnit(enemyIndex)
        enemyIndex = ForceUnitGetNext(enemyIndex)
    end
end

function UnitMoveStepByStep(unit, targetX, targetZ, rotateDirection)
    local startX = UnitGetX(unit)
    local startZ = UnitGetZ(unit)

    -- 如果已经在目标位置，直接返回true
    if startX == targetX and startZ == targetZ then
        return true
    end

    -- 使用BFS寻找路径
    local path = FindPath(unit, startX, startZ, targetX, targetZ)

    -- 如果找不到路径，返回false
    if not path then
        return false
    end

    -- 按路径一步一步移动
    for i = 1, #path do
        local step = path[i]
        UnitMovePos(unit, step.x, step.z, MOVE_FLAG_NONE)
        UnitMoveWait() -- 等待移动完成
    end

    if rotateDirection then
        UnitRotation(unit, rotateDirection)
    end

    return true
end

function FindPath(unit, startX, startZ, targetX, targetZ)
    -- 如果起点和终点相同，直接返回空路径
    if startX == targetX and startZ == targetZ then
        return {}
    end

    -- 初始化BFS
    local queue = {{x = startX, z = startZ, path = {}}}
    local visited = {}
    visited[startX .. "," .. startZ] = true

    -- 四个方向：上、右、下、左
    local directions = {{0, 1}, {1, 0}, {0, -1}, {-1, 0}}

    -- BFS搜索
    while #queue > 0 do
        local current = table.remove(queue, 1)
        local currentX, currentZ = current.x, current.z
        local currentPath = current.path

        -- 检查四个方向
        for _, dir in ipairs(directions) do
            local nextX = currentX + dir[1]
            local nextZ = currentZ + dir[2]
            local nextPosKey = nextX .. "," .. nextZ
            Log(nextPosKey)

            -- 检查边界（假设全局变量g_map_width和g_map_height存在）
            if nextX > 0 and nextX <= g_map_width and nextZ > 0 and nextZ <= g_map_height then
                -- 检查是否已访问过
                if not visited[nextPosKey] then
                    -- 检查是否可以通过该位置
                    if CanPassThroughPosition(unit, nextX, nextZ) then
                        -- 检查是否到达目标
                        if nextX == targetX and nextZ == targetZ then
                            local newPath = {}
                            for i, pos in ipairs(currentPath) do
                                newPath[i] = pos
                            end
                            table.insert(newPath, {x = nextX, z = nextZ})
                            return newPath
                        end

                        -- 添加到队列中继续搜索
                        local newPath = {}
                        for i, pos in ipairs(currentPath) do
                            newPath[i] = pos
                        end
                        table.insert(newPath, {x = nextX, z = nextZ})

                        table.insert(queue, {x = nextX, z = nextZ, path = newPath})
                        visited[nextPosKey] = true
                    end
                end
            end
        end
    end

    -- 如果找不到路径，返回nil
    return nil
end



function CanPassThroughPosition(unit, x, z)
    -- 检查地形是否可通过
    local terrainCost = TerrainGetMoveCost(x, z)
    if terrainCost == "COST_不可" then
        return false
    end

    if terrainCost == "COST_空" and UnitGetMoveCost( unit ) ~= "COST_飛行" then
        return false
    end

    -- 检查是否有单位在该位置
    local unitAtPos = UnitGetByPos(x, z)
    if unitAtPos ~= nil then
        -- 如果是同势力单位，不能通行
        local unitForce = UnitGetForce(unit)
        local targetForce = UnitGetForce(unitAtPos)

        -- 不同势力的单位阻挡路径
        if unitForce ~= targetForce then
            if not UnitHasWholeSkill(unit, "SID_すり抜け") then
                return false
            end
        end
    end

    return true
end

function FindNearestEnemyUnit(targetPid)
    local targetUnit = UnitGetByPID(targetPid)
    if targetUnit == nil then
        return nil
    end

    local targetX = UnitGetX(targetUnit)
    local targetZ = UnitGetZ(targetUnit)

    local nearestEnemy = nil
    local minDistance = 999
    local nearestX = 0
    local nearestZ = 0

    local enemyIndex = ForceUnitGetFirst(FORCE_ENEMY)
    while enemyIndex ~= nil do
        if enemyIndex ~= targetUnit then
            local enemyX = UnitGetX(enemyIndex)
            local enemyZ = UnitGetZ(enemyIndex)

            local distance = math.abs(targetX - enemyX) + math.abs(targetZ - enemyZ)

            if distance < minDistance then
                minDistance = distance
                nearestEnemy = enemyIndex
                nearestX = enemyX
                nearestZ = enemyZ
            end
        end

        enemyIndex = ForceUnitGetNext(enemyIndex)
    end

    return nearestEnemy, nearestX, nearestZ
end

function CalculatePlayerUnitsCenter()
    local totalX = 0
    local totalZ = 0
    local unitCount = 0

    -- 遍历所有我方单位
    local index = ForceUnitGetFirst(FORCE_PLAYER)
    while index ~= nil do
        totalX = totalX + UnitGetX(index)
        totalZ = totalZ + UnitGetZ(index)
        unitCount = unitCount + 1
        index = ForceUnitGetNext(index)
    end

    if unitCount == 0 then
        return 0, 0, 0
    end

    local centerX = totalX / unitCount
    local centerZ = totalZ / unitCount

    return centerX, centerZ, unitCount
end

function TableToString(tbl)
    if type(tbl) ~= "table" then
        return tostring(tbl)
    end

    local result = "{"
    local first = true

    for i, v in ipairs(tbl) do
        if not first then
            result = result .. ","
        end
        first = false

        if type(v) == "table" then
            result = result .. "{"
            local subFirst = true

            for j, subValue in ipairs(v) do
                if not subFirst then
                    result = result .. ","
                end
                subFirst = false

                if type(subValue) == "string" then
                    result = result .. "[" .. j .. "]=" .. '"' .. subValue .. '"'
                else
                    result = result .. "[" .. j .. "]=" .. tostring(subValue)
                end
            end

            for key, subValue in pairs(v) do
                if type(key) ~= "number" or key <= 0 or key > #v or math.floor(key) ~= key then
                    if not subFirst then
                        result = result .. ","
                    end
                    subFirst = false

                    if type(key) == "string" then
                        result = result .. '["' .. key .. '"]='
                    else
                        result = result .. "[" .. tostring(key) .. "]="
                    end

                    if type(subValue) == "string" then
                        result = result .. '"' .. subValue .. '"'
                    else
                        result = result .. tostring(subValue)
                    end
                end
            end

            result = result .. "}"
        elseif type(v) == "string" then
            result = result .. '"' .. v .. '"'
        else
            result = result .. tostring(v)
        end
    end

    result = result .. "}"
    return result
end

function StringToTable(str)
    if str == nil or str == "" then
        return {}
    end

    local func, err = load("return " .. str)
    if func then
        local success, result = pcall(func)
        if success and type(result) == "table" then
            return result
        else
            Log("StringToTable error: " .. tostring(result))
            return {}
        end
    else
        Log("StringToTable load error: " .. tostring(err))
        return {}
    end
end

function splitStrToList(str, delimiter)
    local result = {}
    local from = 1
    local len = string.len(str)
    local delimiter_len = string.len(delimiter)

    for i = 1, len do
        if string.sub(str, i, i + delimiter_len - 1) == delimiter then
            table.insert(result, string.sub(str, from, i - 1))
            from = i + delimiter_len
            i = i + delimiter_len - 1
        end
    end

    table.insert(result, string.sub(str, from))
    return result
end
