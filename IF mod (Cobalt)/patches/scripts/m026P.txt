function SetPlayerGodTable()
	UnitClearPrivateSkill("PID_リュール", "SID_不動_隠蔽")
    local index = ForceUnitGetFirst(FORCE_PLAYER)

    while index ~= nil do
        local unitPid = UnitGetPID(index)
        local equippedRing = UnitGetGodUnit(index)
        if equippedRing ~= nil then
            table.insert(player_God_Table, {unitPid,equippedRing})
		else
            table.insert(player_God_Table, {unitPid,"EmptyGod"})
        end
        index = ForceUnitGetNext(index)
    end

    local playerGodTableStr = TableToString(player_God_Table)
    VariableSet("player_God_Table", playerGodTableStr)

end

end

function ProcessMaidAITargets(maid_God_Table, current_Player_God_Table, valueIndex)
    local index = ForceUnitGetFirst(FORCE_ENEMY)
    while index ~= nil do
        local pid = UnitGetPID(index)
        if string.sub(pid, 1, string.len(maid_Name_Prefix)) == maid_Name_Prefix then
            local targetGid = string.sub(pid, string.len(maid_Name_Prefix) + 1)
            
            local targetValues = FindTargetValues(maid_God_Table, targetGid)
            
            if targetValues ~= nil and #targetValues >= valueIndex then
                local targetPerson = FindMatchingPlayer(current_Player_God_Table, targetValues[valueIndex])
                
                if targetPerson ~= nil then
                    RemoveEntryFromTable(maid_God_Table, targetGid)
                    RemoveEntryFromTable(current_Player_God_Table, targetPerson)
                    
                    AiSetSequence(maid_Name_Prefix .. targetGid, AI_ORDER_ATTACK, "AI_AT_InterferencePerson", targetPerson)
                end
            end
        end
        index = ForceUnitGetNext(index)
    end
end

function ProcessRemainingMaids(maid_God_Table, current_Player_God_Table)
    local emptyGodCount, nonEmptyGodCount = CountPlayerGodTypes(current_Player_God_Table)
    
    local index = ForceUnitGetFirst(FORCE_ENEMY)
    while index ~= nil do
        local pid = UnitGetPID(index)
        if string.sub(pid, 1, string.len(maid_Name_Prefix)) == maid_Name_Prefix then
            local targetGid = string.sub(pid, string.len(maid_Name_Prefix) + 1)
            
            if IsMaidExists(maid_God_Table, targetGid) then
                local targetPerson = nil
                
                if nonEmptyGodCount > 0 then
                    targetPerson, nonEmptyGodCount = SelectNonEmptyGodTarget(current_Player_God_Table, nonEmptyGodCount)
                end
                
                if targetPerson == nil and emptyGodCount > 0 then
                    targetPerson, emptyGodCount = SelectEmptyGodTarget(current_Player_God_Table, emptyGodCount)
                end
                
                if targetPerson ~= nil then
                    AiSetSequence(maid_Name_Prefix .. targetGid, AI_ORDER_ATTACK, "AI_AT_InterferencePerson", targetPerson)
                    RemoveEntryFromTable(maid_God_Table, targetGid)
                end
            end
        end
        index = ForceUnitGetNext(index)
    end
end

function FindTargetValues(maid_God_Table, targetGid)
    for _, entry in ipairs(maid_God_Table) do
        if entry[1] == targetGid then
            return entry[2]
        end
    end
    return nil
end

function FindMatchingPlayer(current_Player_God_Table, targetValue)
    for _, playerEntry in ipairs(current_Player_God_Table) do
        local playerRing = playerEntry[2]
        if playerRing ~= "EmptyGod" then
            local ringName = string.sub(playerRing, 5)
            if ringName == targetValue then
                return playerEntry[1]
            end
        end
    end
    return nil
end

function RemoveEntryFromTable(tableData, searchValue)
    for i, entry in ipairs(tableData) do
        if entry[1] == searchValue then
            table.remove(tableData, i)
            break
        end
    end
end

function CountPlayerGodTypes(current_Player_God_Table)
    local emptyGodCount = 0
    local nonEmptyGodCount = 0
    
    for _, entry in ipairs(current_Player_God_Table) do
        if entry[2] == "EmptyGod" then
            emptyGodCount = emptyGodCount + 1
        else
            nonEmptyGodCount = nonEmptyGodCount + 1
        end
    end
    
    return emptyGodCount, nonEmptyGodCount
end

function IsMaidExists(maid_God_Table, targetGid)
    for _, entry in ipairs(maid_God_Table) do
        if entry[1] == targetGid then
            return true
        end
    end
    return false
end

function SelectNonEmptyGodTarget(current_Player_God_Table, nonEmptyGodCount)
    for i, playerEntry in ipairs(current_Player_God_Table) do
        if playerEntry[2] ~= "EmptyGod" then
            local targetPerson = playerEntry[1]
            table.remove(current_Player_God_Table, i)
            return targetPerson, nonEmptyGodCount - 1
        end
    end
    return nil, nonEmptyGodCount
end

function SelectEmptyGodTarget(current_Player_God_Table, emptyGodCount)
    for i, playerEntry in ipairs(current_Player_God_Table) do
        if playerEntry[2] == "EmptyGod" then
            local targetPerson = playerEntry[1]
            table.remove(current_Player_God_Table, i)
            return targetPerson, emptyGodCount - 1
        end
    end
    return nil, emptyGodCount
end

function ProcessTerrainWithExclusion(processFunc)
    local excludeMap = {}
    for _, excludePoint in ipairs(convert_Exclude_Point) do
        local x, z = excludePoint[1], excludePoint[2]
        if not excludeMap[x] then
            excludeMap[x] = {}
        end
        excludeMap[x][z] = true
    end
    
    TerrainSetBegin()
    processFunc(excludeMap)
    TerrainSetEnd()
end

function IsMaidSummonDirectionPoint(x, z)
    for _, summonInfo in ipairs(maid_Positon_Summon_Direction) do
        local maidX, maidZ = summonInfo[1], summonInfo[2]
        local directionIndex = summonInfo[3]
        local direction = Directions[directionIndex]
        local targetX = maidX + direction[1]
        local targetZ = maidZ + direction[2]
        
        if x == targetX and z == targetZ then
            return true
        end
    end
    return false
end

function InitMapM026P()
    SetMagicBarrierTerrain()
    ConvertAllTerrainWithBarrier()
end

function SetMagicBarrierTerrain()
    ProcessTerrainWithExclusion(function(excludeMap)
        local barrierPoints = magic_Barrier_Point
        
        for i, point in ipairs(barrierPoints) do
            local x = point[1]
            local z = point[2]
            
            local originalTerrain = TerrainGet(x, z)
            table.insert(magic_Barrier_Point_Terra, {x, z, originalTerrain})
            
            TerrainSet(x, z, "TID_魔法障壁")
        end
    end)
    
    local barrierPointTerraStr = TableToString(magic_Barrier_Point_Terra)
    VariableSet("magic_Barrier_Point_Terra", barrierPointTerraStr)
end

function ResetMagicBarrierPoint()
    local barrierPointTerraStr = VariableGet("magic_Barrier_Point_Terra")
    if barrierPointTerraStr == "" then
        return
    end
    
    local savedTerrainData = StringToTable(barrierPointTerraStr)
    
    ProcessTerrainWithExclusion(function(excludeMap)
        for i, terrainInfo in ipairs(savedTerrainData) do
            if type(terrainInfo) == "table" and #terrainInfo >= 3 then
                local x = terrainInfo[1]
                local z = terrainInfo[2]
                local originalTerrain = terrainInfo[3]
                TerrainSet(x, z, originalTerrain)
            end
        end
    end)
    
    VariableSet("magic_Barrier_Point_Terra", "")
    magic_Barrier_Point_Terra = {}
end

function ConvertAllTerrainWithBarrier()
    ProcessTerrainWithExclusion(function(excludeMap)
        for x = 1, g_map_width do
            for z = 1, g_map_height do
                if not (excludeMap[x] and excludeMap[x][z]) then
                    local currentTerrain = TerrainGet(x, z)
                    for _, baseTerrain in ipairs(convert_Terrain_Type) do
                        if currentTerrain == baseTerrain then
                            TerrainSet(x, z, baseTerrain .. "_Barrier_M026P")
                            break
                        end
                    end
                end
            end
        end
    end)
end

function RevertConvertedTerrainOnlyAtMaidSummonDirection()
    ProcessTerrainWithExclusion(function(excludeMap)
        for _, summonInfo in ipairs(maid_Positon_Summon_Direction) do
            local maidX, maidZ = summonInfo[1], summonInfo[2]
            local directionIndex = summonInfo[3]
            
            local direction = Directions[directionIndex]
            local targetX = maidX + direction[1]
            local targetZ = maidZ + direction[2]
            
            if targetX >= 1 and targetX <= g_map_width and targetZ >= 1 and targetZ <= g_map_height then
                if not (excludeMap[targetX] and excludeMap[targetX][targetZ]) then
                    local currentTerrain = TerrainGet(targetX, targetZ)
                    
                    for _, baseTerrain in ipairs(convert_Terrain_Type) do
                        local convertedTerrain = baseTerrain .. "_Barrier_M026P"
                        
                        if currentTerrain == convertedTerrain then
                            TerrainSet(targetX, targetZ, baseTerrain)
                            break
                        end
                    end
                end
            end
        end
        
        for x = 1, g_map_width do
            for z = 1, g_map_height do
                if not (excludeMap[x] and excludeMap[x][z]) then
                    if not IsMaidSummonDirectionPoint(x, z) then
                        local currentTerrain = TerrainGet(x, z)
                        
                        for _, baseTerrain in ipairs(convert_Terrain_Type) do
                            local barrierTerrain = baseTerrain .. "_Barrier_M026P"
                            local banTerrain = baseTerrain .. "_Ban_M026P"
                            
                            if currentTerrain == barrierTerrain then
                                TerrainSet(x, z, banTerrain)
                                break
                            end
                        end
                    end
                end
            end
        end
    end)
end

function RevertBanTerrainToNormal()
    local index = ForceUnitGetFirst(FORCE_ENEMY)
    while index ~= nil do
        UnitClearStatus(index,UNIT_STATUS_MOVE_NOT_ALLOW)
        index = ForceUnitGetNext(index)
    end

    ProcessTerrainWithExclusion(function(excludeMap)
        for x = 1, g_map_width do
            for z = 1, g_map_height do
                if not (excludeMap[x] and excludeMap[x][z]) then
                    local currentTerrain = TerrainGet(x, z)
                    
                    for _, baseTerrain in ipairs(convert_Terrain_Type) do
                        local banTerrain = baseTerrain .. "_Ban_M026P"
                        
                        if currentTerrain == banTerrain then
                            TerrainSet(x, z, baseTerrain)
                            break
                        end
                    end
                end
            end
        end
    end)
end

function FindNearestValidPosition(startX, startY)
    local queue = {}
    local visited = {}
    local directions = {{0, 1}, {1, 0}, {0, -1}, {-1, 0}}

    table.insert(queue, {x = startX, y = startY})
    visited[startX .. "," .. startY] = true

    while #queue > 0 do
        local current = table.remove(queue, 1)
        if IsPositionValid(current.x, current.y) then
            return current.x, current.y
        end
        for _, dir in ipairs(directions) do
            local newX = current.x + dir[1]
            local newY = current.y + dir[2]
            local key = newX .. "," .. newY
            if not visited[key] and IsWithinBounds(newX, newY) and IsPassable(newX, newY) then
                table.insert(queue, {x = newX, y = newY})
                visited[key] = true
            end
        end
    end

    return nil, nil
end

function IsPassable(x, y)
    local terrainCost = TerrainGetMoveCost(x, y)
    return terrainCost != "COST_不可" and terrainCost != "COST_空"
end

function IsWithinBounds(x, y)
    return x >= 1 and x <= g_map_width and y >= 1 and y <= g_map_height
end

function IsPositionValid(x, y)
    local unit = UnitGetByPos(x, y)
    if unit ~= nil then
        return false
    end
    return IsPassable(x, y)
end
