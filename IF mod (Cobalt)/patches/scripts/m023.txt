function Set_Interference_AI_M023()
    for i = 1, 4 do
        local maidPid = "PID_IFMOD_M023_メイド" .. i
        if IsExistForcePlayerNearby(maidPid, 10) == false then
            UnitSetStatus(maidPid, UNIT_STATUS_FIXED)
        end
    end
end


function CheckReinforcementCondition_1()
	if VariableGet("reinforcement_1_triggered") == 1 then
		return false
	end



	local index = ForceUnitGetFirst(FORCE_PLAYER)
	while index ~= nil do
		local z = UnitGetZ(index)
		if (z >= 14 and x <= 24) or MapGetTurn() >= 5 then
			VariableSet("reinforcement_1_triggered", 1)
			return true
		end
		index = ForceUnitGetNext(index)
	end
	return false
end

function InitMapM023()
    TerrainSetBegin()
    for _,point in ipairs(g_rockPoint) do
        local x = point[1]
        local z = point[2]
        TerrainSet( x, z, "TID_平地")
    end
    TerrainSetEnd()

    local magicCirclePoints = {}
    for _, config in ipairs(g_dragonVeinConfig_M023) do
        table.insert(magicCirclePoints, config.point)
    end
    DragonVeinCreateMagicCircle(magicCirclePoints)

    local terrainPoints = {}
    for _, config in ipairs(g_dragonVeinConfig_M023) do
        table.insert(terrainPoints, {config.point[1], config.point[2], "TID_龙脉_M023_SkySeal"})
    end
    DragonVeinSetTerrain(terrainPoints)
end

function EquipDragonVeinSkillsM023()
    local skillIds = {}
    for _, config in ipairs(g_dragonVeinConfig_M023) do
        table.insert(skillIds, config.skillId)
    end
    DragonVeinEquipSkills(skillIds)
end


function DragonVeinUsed()
    local unit = MindGetUnit()
    local x = UnitGetX(unit)
    local z = UnitGetZ(unit)

    local dragonVeinPoints = {}
    for _, config in ipairs(g_dragonVeinConfig_M023) do
        table.insert(dragonVeinPoints, {
            config.point,
            config.skillId,
            config.type,
            config.param
        })
    end

    local veinInfo = DragonVeinPreUseCheck(
        dragonVeinPoints,
        x,
        z
    )

    if veinInfo then
        DragonVeinPostUseHandleM023(veinInfo)
    end
end

function DragonVeinPostUseHandleM023(veinInfo)
    TerrainSetBegin()
        TerrainSet(veinInfo.x, veinInfo.z, "TID_床")
    TerrainSetEnd()

    EffectDelete("魔法陣_G001", veinInfo.x, veinInfo.z)
    CursorSetPos(veinInfo.x, veinInfo.z)
    WaitTime(1.0)
    EffectPlay("水球_発動", veinInfo.x, veinInfo.z)
    WaitTime(1.5)

    for _, config in ipairs(g_dragonVeinConfig_M023) do
        if veinInfo.x == config.point[1] and veinInfo.z == config.point[2] then
            local distOrg = CursorGetDistanceMode()

            CursorSetDistanceMode(CURSOR_DISTANCE_MIDDLE)
            MapCameraWait()

            CursorSetPos((g_map_width+1)/2, (g_map_height+1)/2)
            WaitTime(1.0)
            EffectPlay("水球", (g_map_width+1)/2, (g_map_height+1)/2)
            WaitTime(1.5)

            CursorSetDistanceMode(distOrg)
            MapCameraWait()

            DragonVeinApplyEffect()
            VariableSet("dragonvein_turn_counter",2)

            break
        end
    end
end

function DragonVeinUsedByEnemy()
    if UnitExistOnMap(g_pid_sepia) == false then
        return
    end

    local dragonveinCounter = VariableGet("dragonvein_turn_counter")
    dragonveinCounter = dragonveinCounter - 1

    if dragonveinCounter == 0 then
        CursorSetPos(g_sepia_x,g_sepia_z)
        MapCameraWait()

        EffectPlay( "邪竜紋_発動" , g_sepia_x, g_sepia_z )
        WaitTime( 0.5 )
        dragonveinCounter = 2

        local distOrg = CursorGetDistanceMode()

        CursorSetDistanceMode(CURSOR_DISTANCE_MIDDLE)
        MapCameraWait()

        CursorSetPos((g_map_width+1)/2, (g_map_height+1)/2)
        WaitTime(1.0)
        EffectPlay("カムイ竜脈", (g_map_width+1)/2, (g_map_height+1)/2)
        WaitTime(1.5)

        CursorSetDistanceMode(distOrg)
        MapCameraWait()

        EnemyDragonVeinApplyEffect()
    end

    VariableSet("dragonvein_turn_counter",dragonveinCounter)
end

function CheckStatusLueur()
    if UnitHasPrivateSkill(g_pid_lueur, "SID_移動半減_効果_M023") then
       return -1
    end
    if UnitHasPrivateSkill(g_pid_lueur, "SID_移動半分に増加_効果_M023") then
       return 1
    end
    return 0
end

end

end
