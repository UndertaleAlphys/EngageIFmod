function disposDragonVeinEnemy()
    EffectCreate("魔法陣",19,11)
    TerrainSetBegin()
        TerrainSet(19,11,"TID_龙脉_M019_E")
    TerrainSetEnd()
end

function deleteDragonVeinEnmey()
    EffectDelete("魔法陣",19,11)
    TerrainSetBegin()
        TerrainSet(19,11,"TID_床")
    TerrainSetEnd()
end

function initMapM019()
    disposDragonVeinEnemy()
    ClearAllMiasma()
    local magicCirclePoints = {}
    for _, config in ipairs(g_dragonVeinConfig) do
        table.insert(magicCirclePoints, config.point)
    end
    DragonVeinCreateMagicCircle(magicCirclePoints)

    local terrainPoints = {}
    for _, config in ipairs(g_dragonVeinConfig) do
        table.insert(terrainPoints, {config.point[1], config.point[2], "TID_龙脉_M019_P"})
    end
    DragonVeinSetTerrain(terrainPoints)

    TerrainSetBegin()
    for i, effect in ipairs(summonVortexEffects) do
        EffectCreateRotateXYZ("瘴気", effect[1], effect[2], effect[3], effect[4], effect[5], effect[6], effect[7])
        local currentTerrain = TerrainGet(effect[1],effect[2])
        local newTerrain = currentTerrain .. "_M019"
        TerrainSet(effect[1],effect[2],newTerrain)
    end
    TerrainSet(24,11,"TID_王座")
    TerrainSet(19,11,"TID_王座")
    TerrainSetEnd()
end

function DragonVeinUsed()
    local unit = MindGetUnit()
    local x = UnitGetX(unit)
    local z = UnitGetZ(unit)

    local dragonVeinPoints = {}
    for _, config in ipairs(g_dragonVeinConfig) do
        local firstTargetPoint = config.targetPoints[1] or {0, 0}
        table.insert(dragonVeinPoints, {
            config.point,
            config.skillId,
            "bomb",
            0,
            firstTargetPoint
        })
    end

    local checkX, checkZ = 0, 0

    for _, config in ipairs(g_dragonVeinConfig) do
        if config.point[1] == x and config.point[2] == z then
            checkX = config.targetPoints[1][1]
            checkZ = config.targetPoints[1][2]
            break
        end
    end

    local veinInfo = DragonVeinPreUseCheckWithTid(
        dragonVeinPoints,
        "TID_DragonVein_Used",
        x,
        z,
        checkX,
        checkZ,
        g_map_height,
        g_map_width
    )

    if veinInfo then
        DragonVeinPostUseHandleM019(veinInfo)
    end
end


function DragonVeinPostUseHandleM019(veinInfo)
    TerrainSetBegin()
        TerrainSet(veinInfo.x, veinInfo.z, "TID_床")
    EffectDelete("魔法陣_G001", veinInfo.x, veinInfo.z)
    CursorSetPos(veinInfo.x, veinInfo.z)
    WaitTime(1.0)
    EffectPlay("水球_発動", veinInfo.x, veinInfo.z)
    EffectWait()

    for _, config in ipairs(g_dragonVeinConfig) do
        if veinInfo.x == config.triggerX and veinInfo.z == config.triggerZ then
            CursorSetPos(config.effectX, config.effectZ)
            WaitTime(1.0)
            EffectPlay("カムイ竜脈", config.effectX, config.effectZ)
            EffectWait()
            for _, point in ipairs(config.targetPoints) do
                TerrainSet(point[1], point[2], "TID_床")
                EffectDelete("瘴気", point[1], point[2])
            end

            break
        end
    end

    TerrainSetEnd()
end

function RestoreDeletedEffects()
    if UnitExistOnMap(g_pid_Mauve) == false then
        return
    end
    
    local x = UnitGetX(g_pid_Mauve)
    local z = UnitGetZ(g_pid_Mauve)
    CursorSetPos(x,z)
    WaitTime(1.0)
    EffectPlay( "邪竜紋_発動", x, z )
    WaitTime( 1.5 )
    EffectWait()

    CursorSetPos(g_map_width/2, g_map_height/2)
    WaitTime(1.0)
    EffectPlay("カムイ竜脈", g_map_width/2, g_map_height/2)
    EffectWait()

    TerrainSetBegin()
    for _, effect in ipairs(summonVortexEffects) do
        local x = effect[1]
        local z = effect[2]

        local currentTerrain = TerrainGet(x, z)

        if string.sub(currentTerrain, -5) ~= "_M019" then
            EffectCreateRotateXYZ("瘴気", x, z, effect[3], effect[4], effect[5], effect[6], effect[7])

            local newTerrain = currentTerrain .. "_M019"
            TerrainSet(x, z, newTerrain)
        end
    end
    TerrainSetEnd()
    rebuildDragonVeinInfo()
    ProcessAllUnitsLanding(g_map_width,g_map_height)
    deleteDragonVeinEnmey()
    ReinforcementEventFinal()
end

function rebuildDragonVeinInfo()
    local skillIds = {}
    local dragonVeinPoints = {}
    local magicCirclePoints = {}
    local terrainPoints = {}

    for _, config in ipairs(g_dragonVeinConfig) do
        local x, z = config.point[1], config.point[2]
        if TerrainGet(x, z) ~= "TID_龙脉_M019_P" then
            table.insert(skillIds, config.skillId)
            table.insert(dragonVeinPoints, config.point)
            table.insert(magicCirclePoints, config.point)
            table.insert(terrainPoints, {x, z, "TID_龙脉_M019_P"})
        end
    end

    DragonVeinEquipSkills(skillIds)
    DragonVeinRegisterEvents(dragonVeinPoints)
    DragonVeinCreateMagicCircle(magicCirclePoints)
    DragonVeinSetTerrain(terrainPoints)
end


function ClearAllMiasma()
	TerrainSetBegin()
	for i = 1, g_map_width do
		for j = 1, g_map_height do
			if ( MapOverlapGet( i, j ) == "TID_瘴気_永続") then

				MapOverlapSet( i , j, "TID_無し" )
			end
		end
	end
	MapOverlapSetEnd()
end

function 離脱イベント()
    local index = ForceUnitGetFirst(FORCE_PLAYER)
    while index ~= nil do
        for _, config in ipairs(g_dragonVeinConfig) do
            UnitClearPrivateSkill(index, config.skillId)
        end
        index = ForceUnitGetNext(index)
    end

	local unit = MindGetUnit()
	if not UnitExistOnMap( unit ) then
		return
	end
	
	UnitRotation( g_pid_lueur, ROTATE_UP)
	UnitMoveWait()
	
	VariableSet( "勝利", 1 )
end

function CheckPlayerUnitsPosition()
	if VariableGet("reinforcement_final_triggered") == 1 then
		return false
	end

	local index = ForceUnitGetFirst(FORCE_PLAYER)
	while index ~= nil do
		local x = UnitGetX(index)
		if x >= 19 then
			VariableSet("reinforcement_final_triggered", 1)
			return true
		end
		index = ForceUnitGetNext(index)
	end
	return false
end

function Condition_ReinforcementFinal()
	if VariableGet("reinforcement_final_counter") == 0 then
		return false
	end
	
	local counter = VariableGet("reinforcement_final_counter")
	counter = counter - 1
	VariableSet("reinforcement_final_counter", counter)
	
	if counter == 0 then
		return true
	end
	
	return false
end
