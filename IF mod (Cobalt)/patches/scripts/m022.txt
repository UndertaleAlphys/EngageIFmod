function フラグ登録()

function ProcessUnitList()
    if #mind_unit_list ~= 0 then
        local unit = table.remove(mind_unit_list, 1)
        UnitSetEngageCount(unit,7)
    end
end

function ProcessEnemyDie()
    local godName = string.sub(UnitGetPID(MindGetEventUnit()), #god_prefix + 1)
    local pid = ""
    if UnitGetForce(MindGetUnit()) == FORCE_PLAYER then
        pid = UnitGetPID(MindGetUnit())
    else
        pid = UnitGetPID(MindGetTargetUnit())
    end
    VariableSet("defeat" .. godName,1)
    local defeatCount = VariableGet("DefeatedGodCount")
    VariableSet("DefeatedGodCount",defeatCount+1)

    local isEngaging = UnitGetEngaging(MindGetUnit())
    local engageTurn = UnitGetEngageTurn(MindGetUnit())

    if godName ~= "エイリーク" and godName ~= "エフラム" and godName ~= "クロード" and godName ~= "ディミトリ" and godName ~= "エーデルガルト" and godName ~= "ルフレ" and godName ~= "クロム"then
        EffectPlay( "竜化_G1" , UnitGetX(pid), UnitGetZ(pid) )
        UnitSetEngageCount(MindGetUnit(),7)
        WaitTime( 0.5 )
        GodUnitSetEscape("GID_" .. godName,	false)
        UnitSetGodUnit(pid,"GID_" .. godName)
        WaitTime( 0.5 )
        UnitSetEngageCount(MindGetUnit(),7)
    end

    if godName == "クロード" or godName == "ディミトリ" or godName == "エーデルガルト" then
        if VariableGet("defeat" .. "クロード" ) == 1 and VariableGet("defeat" .. "ディミトリ" ) == 1 and VariableGet("defeat" .. "エーデルガルト" ) == 1 then
            EffectPlay( "竜化_G1" , UnitGetX(pid), UnitGetZ(pid) )
            UnitSetEngageCount(MindGetUnit(),7)
            WaitTime( 0.5 )
            GodUnitSetEscape("GID_エーデルガルト",	false)
            UnitSetGodUnit(pid,"GID_エーデルガルト")
            WaitTime( 0.5 )
        end
    end

    if godName == "ルフレ" or godName == "クロム"  then
        if VariableGet("defeat" .. "ルフレ" ) == 1 and VariableGet("defeat" .. "クロム" ) == 1 then
            EffectPlay( "竜化_G1" , UnitGetX(pid), UnitGetZ(pid) )
            UnitSetEngageCount(MindGetUnit(),7)
            WaitTime( 0.5 )
            GodUnitSetEscape("GID_クロム",	false)
            UnitSetGodUnit(pid,"GID_クロム")
            WaitTime( 0.5 )
        end
    end

     if godName == "エイリーク" or godName == "エフラム"  then
        if VariableGet("defeat" .. "エイリーク" ) == 1 and VariableGet("defeat" .. "エフラム" ) == 1 then
            EffectPlay( "竜化_G1" , UnitGetX(pid), UnitGetZ(pid) )
            UnitSetEngageCount(MindGetUnit(),7)
            WaitTime( 0.5 )
            GodUnitSetEscape("エイリーク",	false)
            UnitSetGodUnit(pid,"GID_エイリーク")
            WaitTime( 0.5 )
        end
    end

    table.insert(mind_unit_list,MindGetUnit())

    if isEngaging then
        UnitSetEngaging(MindGetUnit(),true)
        UnitSetEngageTurn(MindGetUnit(),engageTurn)
    end

    if VariableGet("DefeatedGodCount") == 21 then
        VariableSet( "勝利", 1 )
    end
end

function InitMapM022()
    Log("InitMapM022")
    TerrainSetBegin()
    for z = 1, g_map_height do
        for x = 1, g_map_width do
            local tid = TerrainGet(x, z)
            Log(tid)
            if tid and string.sub(tid, -3) == "の指輪" then
                TerrainSet(x, z, "TID_床")
            end
        end
    end
    TerrainSetEnd()
end

function ユニットと紋章士の会話()
	
	local mind_pid		= UnitGetPID( MindGetUnit() )
	local target_pid	= UnitGetPID( MindGetTargetUnit() )
	local gid			= UnitGetGodUnit( mind_pid )
	local target_gid	= 紋章士ID取得( target_pid )
	local target_mgid	= GodDataGetMGID( target_gid )
	
	
	-- 会話
	if gid == nil then
		
		TalkBeginContinue()
			MessSetArgument( 0, target_mgid )
			ユニット会話_ソロ時( mind_pid )
			紋章士会話( target_pid )
		TalkEndContinue()
		
	else
		
		local key = 指輪拾うフラグ生成( mind_pid )
		if ( VariableGet( key ) == 0 ) then
		
			MessSetArgument( 0, target_mgid )
			ユニット会話_シンクロ中( mind_pid )
			VariableSet( key, 1 )
			
		end
		
		Dialog( "MID_TUT_NAVI_M022_GET_" .. SubPrefix( target_mgid ) )
		
	end
	
	
	-- 単体ユニット削除
	if UnitExistOnMap( target_pid ) then
		UnitDelete( target_pid )
	end
	
	
	-- 離脱フラグ解除
	GodUnitSetEscape(target_gid, false)
	
	-- 闇化フラグ設定
	GodUnitSetDarkness( target_gid, true )
	
	
	-- シンクロ
	if gid == nil then
		
		UnitSetGodUnit( mind_pid, target_gid )
	end

end


function VictoryCheck()
	
	CursorSetPos_FromPid( g_pid_lueur )
	Talk( "MID_EV6" )
	
	
	local index = ForceUnitGetFirst(FORCE_ENEMY)
	if index == nil then
		VariableSet( "勝利", 1 )
		UnitSetStatus( MindGetUnit(), UNIT_STATUS_FIXED )
		
	else
		-- 勝利条件の変更
		WinRuleSetEnemyNumberLessThanOrEqualTo(0)
		WinRuleSetMID( "MID_RULE_ANNIHILATE" )
		WinRule()
		
	end
	
	VariableSet( g_key_ring_complete, 1 )
	
end

function 離脱イベント()
	local unit = MindGetUnit()
	if not UnitExistOnMap( unit ) then
		return
	end
	
	UnitRotation( g_pid_lueur, ROTATE_UP)
	UnitMoveWait()
	
	VariableSet( "勝利", 1 )
