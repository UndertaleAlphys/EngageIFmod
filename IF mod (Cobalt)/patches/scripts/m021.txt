function SaveDragonVeinEffectTurns()
    local serializedData = ""
    for pid, turns in pairs(g_dragonVeinEffectTurns) do
        if serializedData ~= "" then
            serializedData = serializedData .. ";"
        end
        serializedData = serializedData .. pid .. "," .. turns
    end
    VariableSet("DragonVeinEffectTurns", serializedData)
end

function SplitString(str, delimiter)
    local result = {}
    local start = 1
    local str_len = string.len(str)
    local delimiter_len = string.len(delimiter)

    function find_delimiter(s, delim, start_pos)
        local s_len = string.len(s)
        local delim_len = string.len(delim)

        for i = start_pos, s_len - delim_len + 1 do
            local found = true
            for j = 1, delim_len do
                if string.sub(s, i + j - 1, i + j - 1) ~= string.sub(delim, j, j) then
                    found = false
                    break
                end
            end
            if found then
                return i
            end
        end
        return nil
    end

    local pos = find_delimiter(str, delimiter, start)

    while pos do
        local substr = string.sub(str, start, pos - 1)
        table.insert(result, substr)
        start = pos + delimiter_len
        pos = find_delimiter(str, delimiter, start)
    end

    local substr = string.sub(str, start)
    table.insert(result, substr)

    return result
end

function LoadDragonVeinEffectTurns()
    g_dragonVeinEffectTurns = {}
    local serializedData = VariableGet("DragonVeinEffectTurns")
    if serializedData ~= "" then
        local entries = SplitString(serializedData, ";")
        for i, entry in ipairs(entries) do
            local parts = SplitString(entry, ",")
            if #parts == 2 then
                local pid = parts[1]
                local turns = tonumber(parts[2])
                g_dragonVeinEffectTurns[pid] = turns
            end
        end
    end
end

function DragonVeinUsed()
    local unit = MindGetUnit()
    local x = UnitGetX(unit)
    local z = UnitGetZ(unit)

    local dragonVeinPoints = {}
    for _, config in ipairs(g_dragonVeinConfig_M021) do
        table.insert(dragonVeinPoints, {
            config.point,
            config.skillId,
            config.type,
            config.param
        })
    end

    local veinInfo = DragonVeinPreUseCheck(
        dragonVeinPoints,
        x,
        z
    )

    if veinInfo then
        DragonVeinPostUseHandleM021(veinInfo)
    end
end

function DragonVeinPostUseHandleM021(veinInfo)
    TerrainSetBegin()
        TerrainSet(veinInfo.x, veinInfo.z, "TID_床")
    TerrainSetEnd()

    EffectDelete("魔法陣_G001", veinInfo.x, veinInfo.z)
    CursorSetPos(veinInfo.x, veinInfo.z)
    WaitTime(1.0)
    EffectPlay("水球_発動", veinInfo.x, veinInfo.z)
    WaitTime(1.5)

    local centerX = veinInfo.x
    local centerZ = veinInfo.z
    local range = 3

    local playerUnitIndex = ForceUnitGetFirst(FORCE_PLAYER)
    while playerUnitIndex ~= nil do
        local unitX = UnitGetX(playerUnitIndex)
        local unitZ = UnitGetZ(playerUnitIndex)

        if math.abs(unitX - centerX) <= range and math.abs(unitZ - centerZ) <= range then
            UnitSetPrivateSkill(playerUnitIndex, "SID_M021_DragonVeinEffect")
            local playerPid = UnitGetPID(playerUnitIndex)
            g_dragonVeinEffectTurns[playerPid] = 2
        end

        playerUnitIndex = ForceUnitGetNext(playerUnitIndex)
    end

    --RollbackUnitEngageStatusByIndex(1,2)

    SaveDragonVeinEffectTurns()
end

function PlayerTurnDragonVeinTurnManage()
    LoadDragonVeinEffectTurns()

    local updated = false
    for pid, turns in pairs(g_dragonVeinEffectTurns) do
        g_dragonVeinEffectTurns[pid] = turns - 1
        updated = true
    end

    local changed = false
    for pid, turns in pairs(g_dragonVeinEffectTurns) do
        if turns <= 0 then
            g_dragonVeinEffectTurns[pid] = nil
            changed = true
        end
    end

    if updated or changed then
        SaveDragonVeinEffectTurns()
    end
end

function IsUnitInEngageSealList(pid)
    for i, listPid in ipairs(UnitListHasEngageSeal) do
        if listPid == pid then
            return true
        end
    end
    return false
end

function GetUnitsInRange(unitX, unitZ, force, maxDistance)
    local unitsInRange = {}
    local unitIndex = ForceUnitGetFirst(force)
    while unitIndex ~= nil do
        local targetX = UnitGetX(unitIndex)
        local targetZ = UnitGetZ(unitIndex)
        local distance = 二点間距離(unitX, unitZ, targetX, targetZ)
        if distance <= maxDistance then
            table.insert(unitsInRange, unitIndex)
        end
        unitIndex = ForceUnitGetNext(unitIndex)
    end
    return unitsInRange
end

function SetEngageSealFlagde()
    local enemyIndex = ForceUnitGetFirst(FORCE_ENEMY)
    while enemyIndex ~= nil do
        local pid = UnitGetPID(enemyIndex)
        if IsUnitInEngageSealList(pid) then
            local enemyX = UnitGetX(enemyIndex)
            local enemyZ = UnitGetZ(enemyIndex)
            local playerUnitsInRange = GetUnitsInRange(enemyX, enemyZ, FORCE_PLAYER, 10)

            local filteredUnits = {}
            for i, playerIndex in ipairs(playerUnitsInRange) do
                local playerPid = UnitGetPID(playerIndex)
                if not g_dragonVeinEffectTurns[playerPid] then
                    table.insert(filteredUnits, playerIndex)
                end
            end

            if #filteredUnits > 0 then
                EffectPlay("邪竜紋_発動", enemyX, enemyZ)
                WaitTime(2.0)
                for i, playerIndex in ipairs(filteredUnits) do
                    local playerX = UnitGetX(playerIndex)
                    local playerZ = UnitGetZ(playerIndex)
                    EffectPlay("エンゲージブレイク_ヒット", playerX, playerZ)
                    WaitTime(0.5)
                    UnitSetPrivateSkill(playerIndex, "SID_IFMOD_EngageSeal_Flagde")

                    local isEngaging = UnitGetEngaging(playerIndex)

                    if isEngaging then
                        local engageTurn = UnitGetEngageTurn(playerIndex)
                        local postEngageTurn = math.max(0,engageTurn+1)
                        UnitSetEngageTurn(playerIndex,postEngageTurn)
                    else
                        local preEngageCount = UnitGetEngageCount(playerIndex)
                        UnitSetEngageCount(playerIndex,math.max(0,preEngageCount-2))
                    end
                end
            end
        end
        enemyIndex = ForceUnitGetNext(enemyIndex)
    end
end


function SetEngageSealRange()
    MapRangeClear()
    local enemyIndex = ForceUnitGetFirst(FORCE_ENEMY)
    while enemyIndex ~= nil do
        local pid = UnitGetPID(enemyIndex)
        if IsUnitInEngageSealList(pid) then
            local enemyX = UnitGetX(enemyIndex)
            local enemyZ = UnitGetZ(enemyIndex)
            MapRangeAddBegin()
            for x = enemyX - 10, enemyX + 10 do
                for z = enemyZ - 10, enemyZ + 10 do
                    if x >= 1 and x <= g_map_width and z >= 1 and z <= g_map_height then
                        local distance = 二点間距離(enemyX, enemyZ, x, z)
                        if distance <= 10 then
                            local moveCost = TerrainGetMoveCost(x, z)
                            if moveCost ~= "COST_不可" then
                                MapRangeAdd(x, z)
                            end
                        end
                    end
                end
            end
            MapRangeAddEnd()
        end
        enemyIndex = ForceUnitGetNext(enemyIndex)
    end
end

function SavePlayerUnitsEngageStatus()
    local playerUnit = ForceUnitGetFirst(FORCE_PLAYER)
    local unitIndex = 1

    while playerUnit ~= nil do
        local engageCount = UnitGetEngageCount(playerUnit)
        local isEngaging = UnitGetEngaging(playerUnit)
        local engageTurn = UnitGetEngageTurn(playerUnit) + 1
        local usedEngageAttack = HasUnitUsedEngageAttack(playerUnit)
        local isEngagingValue = isEngaging and 1 or 0
        local usedEngageAttackValue = usedEngageAttack and 1 or 0

        local unitGlobalCounterName = "EngageStatus_GlobalCounter" .. unitIndex
        local unitGlobalCounter = 0

        if VariableGet(unitGlobalCounterName) == 0 then
            VariableEntry(unitGlobalCounterName, unitGlobalCounter)
        else
            unitGlobalCounter = VariableGet(unitGlobalCounterName)
        end

        if unitGlobalCounter >= 15 then
            for i = 2, 15 do
                local sourceVariableName = "PlayerEngageStatus_" .. unitIndex .. "_" .. i
                local targetVariableName = "PlayerEngageStatus_" .. unitIndex .. "_" .. (i - 1)

                if VariableGet(sourceVariableName) ~= nil and VariableGet(sourceVariableName) ~= 0 then
                    local data = VariableGet(sourceVariableName)
                    VariableSet(targetVariableName, data)
                else
                    VariableSet(targetVariableName, 0)
                end
            end
            unitGlobalCounter = 15
        else
            unitGlobalCounter = unitGlobalCounter + 1
        end

        VariableSet(unitGlobalCounterName, unitGlobalCounter)

        local variableName = "PlayerEngageStatus_" .. unitIndex .. "_" .. unitGlobalCounter
        local unitData = {
            engageCount,
            isEngagingValue,
            engageTurn,
            usedEngageAttackValue,
            unitGlobalCounter,
        }

        local serializedData = SerializeUnitEngageData(unitData)

        if VariableGet(variableName) == 0 then
            VariableEntry(variableName, serializedData)
        else
            VariableSet(variableName, serializedData)
        end

        unitIndex = unitIndex + 1
        playerUnit = ForceUnitGetNext(playerUnit)
    end
end


function SerializeUnitEngageData(unitData)
    local result = ""
    result = result .. tostring(unitData[1]) .. ","
    result = result .. tostring(unitData[2]) .. ","
    result = result .. tostring(unitData[3]) .. ","
    result = result .. tostring(unitData[4]) .. ","
    result = result .. tostring(unitData[5])
    return result
end



function splitStringLua(str, delimiter)
    local result = {}
    local from = 1
    local str_len = string.len(str)
    local delimiter_len = string.len(delimiter)

    if delimiter_len == 0 or str_len == 0 then
        table.insert(result, str)
        return result
    end

    while from <= str_len do
        local pos = findDelimiterPosition(str, delimiter, from)

        if pos then
            table.insert(result, string.sub(str, from, pos - 1))
            from = pos + delimiter_len
        else
            table.insert(result, string.sub(str, from))
            break
        end
    end

    if string.sub(str, -delimiter_len) == delimiter then
        table.insert(result, "")
    end

    return result
end

function findDelimiterPosition(str, delimiter, start_pos)
    local str_len = string.len(str)
    local delimiter_len = string.len(delimiter)

    for i = start_pos, str_len - delimiter_len + 1 do
        local match = true
        for j = 1, delimiter_len do
            if string.sub(str, i + j - 1, i + j - 1) ~= string.sub(delimiter, j, j) then
                match = false
                break
            end
        end
        if match then
            return i
        end
    end

    return nil
end


function DeserializeUnitEngageData(savedData)
    local result = {}
    local fields = splitStringLua(savedData, ",")

    for _, field in ipairs(fields) do
        local value = tonumber(field) or field
        table.insert(result, value)
    end
    return result
end

function GetPlayerEngageStatusByUnitGlobalCounter(unitIndex, unitGlobalCounter)
    local variableName = "PlayerEngageStatus_" .. unitIndex .. "_" .. unitGlobalCounter
    if VariableGet(variableName) ~= nil then
        local savedData = VariableGet(variableName)
        if savedData ~= "" then
            return DeserializeUnitEngageData(savedData)
        end
    end
    return nil
end

function GetLatestPlayerEngageStatusByUnitWithIndex(unitIndex)
    local unitGlobalCounterName = "EngageStatus_GlobalCounter" .. unitIndex
    if VariableGet(unitGlobalCounterName) ~= nil then
        local unitGlobalCounter = VariableGet(unitGlobalCounterName)
        unitGlobalCounter = math.min(unitGlobalCounter, 15)
        return GetPlayerEngageStatusByUnitGlobalCounter(unitIndex, unitGlobalCounter)
    end
    return nil
end

function PrintAllPlayerUnitsEngageStatus()
    local unitIndex = 1
    local unitGlobalCounterName = "EngageStatus_GlobalCounter" .. unitIndex
    if VariableGet(unitGlobalCounterName) ~= nil then
        local currentCounter = VariableGet(unitGlobalCounterName)
        local maxRecords = math.min(currentCounter, 15)

        Log("Unit Index: " .. unitIndex .. " (Printing " .. maxRecords .. " history records)")
        for i = 1, maxRecords do
            local unitData = GetPlayerEngageStatusByUnitGlobalCounter(unitIndex, i)
            if unitData then
                Log("  Record " .. i .. ":")
                Log("    Engage Count: " .. tostring(unitData[1]))
                Log("    Is Engaging: " .. tostring(unitData[2]))
                Log("    Engage Turn: " .. tostring(unitData[3]))
                Log("    Used Engage Attack: " .. tostring(unitData[4]))
                Log("    Global Counter: " .. tostring(unitData[5]))
            else
                Log("  Record " .. i .. ": No Data")
            end
        end
    else
        Log("Unit Index: " .. unitIndex .. " (No History Data)")
    end
end



function RollbackUnitEngageStatusByIndex(unitIndex, rollbackSteps)
    local unitGlobalCounterName = "EngageStatus_GlobalCounter" .. unitIndex
    if VariableGet(unitGlobalCounterName) ~= nil then
        local currentUnitGlobalCounter = VariableGet(unitGlobalCounterName)
        local targetUnitGlobalCounter = math.max(1, math.min(15, currentUnitGlobalCounter - rollbackSteps))
        VariableSet(unitGlobalCounterName, targetUnitGlobalCounter)

        local unitData = GetPlayerEngageStatusByUnitGlobalCounter(unitIndex, targetUnitGlobalCounter)

        if unitData then
            local playerUnit = ForceUnitGetFirst(FORCE_PLAYER)
            while playerUnit ~= nil do
                if playerUnit == unitIndex then
                    if unitData[2] == 1 then
                        UnitSetEngaging(playerUnit, true)
                        UnitSetEngageTurn(playerUnit, unitData[3])
                    else
                        UnitSetEngaging(playerUnit, false)
                        UnitSetEngageCount(playerUnit, unitData[1])
                    end

                    if unitData[4] == 0 then
                        ResetEngageAttackStatus(playerUnit)
                    elseif unitData[4] == 1 and unitData[2] == 1 then
                        SetHasUsedEngageAttack(playerUnit)
                    end

                    break
                end
                playerUnit = ForceUnitGetNext(playerUnit)
            end
        end
    end
end
