function InitTerrain()
    TerrainSetBegin()
    for z = 0, g_height - 1 do
        for x = 0, g_width - 1 do
            if TerrainGet(x,z) == "TID_防衛床_ターゲット・ワープ禁" then
                TerrainSet(x, z, "TID_床")
            end
            if TerrainGet(x,z) == "TID_防衛床" then
                TerrainSet(x, z, "TID_床")
            end
        end
    end
    TerrainSetEnd()
end

function MapOverlapAllClearFixed()

	local num = VariableGet( g_key_BrokenCrystalNum )

	if ( num == g_CrystalMaxNum ) then
		MapOverlapSetBegin()
		MapOverlapAllClear()
		MapOverlapSetEnd()
	end
end

function MapOverlapAllClear()
	for z = 1, g_height-2 do
		for x = 1, g_width-2 do
			local key = x + ( z - 1 ) * ( g_width - 2 )
			if MapOverlapGet( x, z) == "TID_霧_永続"then
				MapOverlapSet(x, z, "TID_無し")
			end
		end
	end
end

function MapGenerateFogByTurn(turn)
	if VariableGet( g_key_BrokenBarrier ) == 1 then
		return
	end
	MapOverlapSetBegin()
		for z = 1, g_height-2 do
			for x = 1, g_width-2 do
				local key = x + ( z - 1 ) * ( g_width - 2 )
				if g_puddleMap[key] == turn then
					MapOverlapSet(x, z, "TID_霧_永続")
				end
			end
		end
	MapOverlapSetEnd()
end

function MapGenerateFog()
	MapOverlapSetBegin()
	for z = 1, g_height-2 do
		for x = 1, g_width-2 do
			local key = x + ( z - 1 ) * ( g_width - 2 )
			if g_puddleMap[key] == 2 then
				MapOverlapSet(x, z, "TID_霧_永続")
			end
		end
	end
	MapOverlapSetEnd()
end


function Fog_Change_Impl()
	if VariableGet( g_key_BrokenBarrier ) == 1 then
		return
	end
	-- CoblatNotice( "Fog Change" )
	local turn = MapGetTurn() % 3 + 1
	MapOverlapSetBegin()
	for z = 1, g_height-2 do
		for x = 1, g_width-2 do
			local key = x + ( z - 1 ) * ( g_width - 2 )
			if g_puddleMap[key] == turn then
				MapOverlapSet(x, z, "TID_霧_永続")
			else
				MapOverlapSet(x, z, "TID_無し")
			end
		end
	end
	MapOverlapSetEnd()
end

function OddEnemyChangeIntoLeaf()
	local turn = MapGetTurn() % 2
	if (turn == 1) then
		-- CoblatNotice( "Odd Enemy Turn into Leaf Status" )
		if #OddPIDs == 0 then
			return
		end

		for i = 1, #OddPIDs do
			local pid = OddPIDs[i]
			local index = ForceUnitGetFirst(FORCE_ENEMY)
			while index ~= nil do
				if UnitGetPID(index) == pid then
					UnitSetPrivateSkill(index,"SID_Leaves")
					AiSetSequence( index, AI_ORDER_ATTACK, "AI_AT_Null", "")
					if (pid == "PID_G006_ルフレ") then
						AiSetSequence( index, AI_ORDER_ATTACK, "AI_AT_CSYell", "")
					end
				end
				index = ForceUnitGetNext(index)
			end
		end
	end
end

function OddEnemySetBarrier()
	local turn = MapGetTurn() % 2
	if (turn == 0) then
		-- CoblatNotice( "Odd Enemy Set Terrain" )
		if #OddPIDs == 0 then
			return
		end

		for i = 1, #OddPIDs do
			local pid = OddPIDs[i]
			local index = ForceUnitGetFirst(FORCE_ENEMY)
			while index ~= nil do
				if UnitGetPID(index) == pid then
					local x = UnitGetX(index)
					local z = UnitGetZ(index)
					if (TerrainGet(x, z) == "TID_床") then
						TerrainSetBegin()
							TerrainSet(x, z, "TID_床_Barrier")
						TerrainSetEnd()
					end
					if (TerrainGet(x, z) == "TID_柱") then
						TerrainSetBegin()
							TerrainSet(x, z, "TID_柱_Barrier")
						TerrainSetEnd()
					end
					AiSetSequence( index, AI_ORDER_ATTACK, "AI_AT_AttackCS", "")
				end
				index = ForceUnitGetNext(index)
			end
		end
	end
end

function EvenEnemySetBarrier()
	local turn = MapGetTurn() % 2
	if (turn == 1) then
		-- CoblatNotice( "Even Enemy Set Terrain" )
		if #EvenPIDs == 0 then
			return
		end

		for i = 1, #EvenPIDs do
			local pid = EvenPIDs[i]
			local index = ForceUnitGetFirst(FORCE_ENEMY)
			while index ~= nil do
				if UnitGetPID(index) == pid then
					local x = UnitGetX(index)
					local z = UnitGetZ(index)
					if (TerrainGet(x, z) == "TID_床") then
						TerrainSetBegin()
							TerrainSet(x, z, "TID_床_Barrier")
						TerrainSetEnd()
					end
					if (TerrainGet(x, z) == "TID_柱") then
						TerrainSetBegin()
							TerrainSet(x, z, "TID_柱_Barrier")
						TerrainSetEnd()
					end
					AiSetSequence( index, AI_ORDER_ATTACK, "AI_AT_AttackCS", "")
					if (pid == "PID_G006_クロム") then
						AiSetSequence( index, AI_ORDER_ATTACK, "AI_AT_EngageAttack", "")

function OddEnemySetBarrierPostProcess()
	local turn = MapGetTurn() % 2
	if (turn == 0) then
		-- CoblatNotice( "Odd Enemy Turn into Normal and Clear Terrain" )
		if #OddPIDs == 0 then
			return
		end

		for i = 1, #OddPIDs do
			local pid = OddPIDs[i]
			local index = ForceUnitGetFirst(FORCE_ENEMY)
			while index ~= nil do
				if UnitGetPID(index) == pid then
					local x = UnitGetX(index)
					local z = UnitGetZ(index)
					if (TerrainGet(x, z) == "TID_床_Barrier") then
						TerrainSetBegin()
							TerrainSet(x, z, "TID_床")
						TerrainSetEnd()
					end
					if (TerrainGet(x, z) == "TID_柱_Barrier") then
						TerrainSetBegin()
							TerrainSet(x, z, "TID_柱")
						TerrainSetEnd()
					end
					UnitClearPrivateSkill( index, "SID_Leaves" )
				end
				index = ForceUnitGetNext(index)
			end
		end
	end
end

function condition_プレイヤー軍ばらつきチェック()
	
	if VariableGet( g_key_puddleOn ) == 1 then
		return false
	end
	
	
	-- ユニット配置のばらつきが大きくなりはじめたらイベント再生
	
	local center_x, center_z = プレイヤー軍の中心点を算出()
	
	local totalS = 0
	local count = 0
	
	local unit = ForceUnitGetFirst(FORCE_PLAYER)
	while unit != nil do
		
		local sx = math.abs( UnitGetX( unit ) - center_x )
		local sz = math.abs( UnitGetZ( unit ) - center_z )
		
		totalS = totalS + sx + sz
		count = count + 1
		
		unit = ForceUnitGetNext(unit)
	end
	
	local aveS = totalS / count
	-- Log( "標準偏差の平均は " .. tostring( aveS ) )
	if aveS >= 5 then
		VariableSet( g_key_puddleOn_Judge, 1 )
	end
	
end
