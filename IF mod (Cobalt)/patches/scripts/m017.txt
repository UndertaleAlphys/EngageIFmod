function setPlayerGodTable()
    local index = ForceUnitGetFirst(FORCE_PLAYER)

    while index ~= nil do
        local unitPid = UnitGetPID(index)
        local equippedRing = UnitGetGodUnit(index)
        if equippedRing ~= nil then
            table.insert(player_God_Table, {
                pid = unitPid,
                ring = equippedRing
            })
        end
        index = ForceUnitGetNext(index)
    end

    for _, godId in ipairs(canUsedGodList) do
        local isEquipped = false

        for _, equippedInfo in ipairs(player_God_Table) do
            if equippedInfo.ring == godId then
                isEquipped = true
                break
            end
        end

        if not isEquipped then
            table.insert(notEquipGodList, godId)
        end
    end
end

function setGodFromUnequip()
    for _, godId in ipairs(notEquipGodList) do
        if not IsGodStolen(godId) then
            StealGodRing(godId,g_pid_lueur)
        end
    end
end

function StealGodRing(godId,fromPid)
    local nearestEnemy, targetX, targetZ = FindNearestEnemyUnitHasNoGod(fromPid)

    if nearestEnemy ~= nil then
        -- 显示效果
        CursorSetPos(targetX, targetZ)
        MapCameraWait()
        EffectPlay("ワープイン_闇", targetX, targetZ)
        UnitShine(nearestEnemy, 0.3)
        WaitTime(0.3)

        -- 为敌方单位装备敌方版本的戒指
        local newGodId = ConvertToEnemyGodId(godId)
        UnitCreateGodUnit(nearestEnemy, newGodId)

        -- 标记该戒指已被夺取
        SetGodStolen(godId)

        CobaltNotice("敌方单位在位置(" .. targetX .. "," .. targetZ .. ")装备了戒指: " .. newGodId)
    end
end

function StealGodRingFromPlayer()
    local stealablePlayerRings = {}

    for _, equippedInfo in ipairs(player_God_Table) do
        local godId = equippedInfo.ring
        -- 检查该戒指是否可偷取且未被偷取过
        if IsGodStealable(godId) and not IsGodStolen(godId) then
            table.insert(stealablePlayerRings, equippedInfo)
        end
    end

    local stealCount = math.min(2, #stealablePlayerRings)
    Log("stealCount" .. stealCount)
    -- 如果没有可偷取的戒指，直接返回
    if stealCount <= 0 then
        return
    end

    local stolenRings = {}
    for i = 1, stealCount do
        local randomIndex = math.random(1, #stealablePlayerRings)
        table.insert(stolenRings, stealablePlayerRings[randomIndex])
        table.remove(stealablePlayerRings, randomIndex)
    end

    for _, equippedInfo in ipairs(stolenRings) do
        local godId = equippedInfo.ring
        local unitPid = equippedInfo.pid

        -- 找到装备该戒指的单位位置
        if UnitExistOnMap(unitPid) then
            local unitX = UnitGetX(unitPid)
            local unitZ = UnitGetZ(unitPid)

            -- 找到最近的未装备戒指的敌方单位
            local nearestEnemy = FindNearestEnemyUnitHasNoGod(unitPid)

            if nearestEnemy ~= nil then
                -- 显示效果
                CursorSetPos(unitX, unitZ)
                MapCameraWait()
                EffectPlay("ワープアウト_闇", unitX, unitZ)
                UnitShine(UnitGetByPID(unitPid), 0.3)
                WaitTime(0.3)

                -- 从玩家单位移除戒指
                UnitSetGodUnit(unitPid, nil)

                -- 为敌方单位装备敌方版本的戒指
                local enemyGodId = ConvertToEnemyGodId(godId)
                UnitCreateGodUnit(nearestEnemy, enemyGodId)

                -- 标记该戒指已被偷取
                SetGodStolen(godId)

                CobaltNotice("从玩家单位 " .. unitPid .. " 处偷取了戒指 " .. godId .. " 并装备给敌方单位")
            end
        end
    end
end

-- 检查被击杀敌方单位的戒指
function checkDeadEnemy()
    Log("step1")
    local deadUnit = MindGetEventUnit()  -- 获取被击杀的单位
    Log("step2")
    if deadUnit == nil then
        return
    end

    local enemyGodId = UnitGetGodUnit(deadUnit)  -- 获取敌方单位装备的戒指
    Log("step3")
    if enemyGodId == nil then
        return  -- 没有装备戒指，直接返回
    end

    -- 将敌方戒指ID转换为我方版本
    local playerGodId = ConvertToNormalGodId(enemyGodId)
    Log("step4")

    if playerGodId == enemyGodId then
        return  -- 转换失败，不是我们需要处理的戒指
    end

    -- 检查该戒指是否在notEquipGodList中
    local isInNotEquipList = false
    for _, godId in ipairs(notEquipGodList) do
        if godId == playerGodId then
            isInNotEquipList = true
            break
        end
    end
    Log("step5")

    if not isInNotEquipList then
        -- 不在notEquipGodList中，搜寻player_God_Table找到原先装备该戒指的单位
        for _, equippedInfo in ipairs(player_God_Table) do
            if equippedInfo.ring == playerGodId then
                -- 找到原先装备该戒指的单位，重新为其装备戒指
                UnitSetGodUnit(equippedInfo.pid, playerGodId)
                SetGodAvailable(playerGodId)
                Log("step5-1")
                break
            end
        end
    else
        -- 在notEquipGodList中，只刷新其偷取状态
        CobaltNotice(playerGodId)
        SetGodAvailable(playerGodId)
        Log("step5-2")
    end
end

function ConvertToEnemyGodId(godId)
    if not IsGodStealable(godId) then
        return godId
    end

    if string.sub(godId, 1, 4) == "GID_" then
        local baseName = string.sub(godId, 5)
        if baseName then
            return "GID_M017_敵" .. baseName
        end
    end
    return godId
end

function ConvertToNormalGodId(enemyGodId)
    Log(enemyGodId)
    if string.sub(enemyGodId, 1, 10) == "GID_M017_敵" then
        local baseName = string.sub(enemyGodId, 11)
        Log(baseName)
        if baseName then
            local normalGodId = "GID_" .. baseName
            Log(normalGodId)
            -- 检查转换后的戒指是否在可夺取列表中
            if IsGodStealable(normalGodId) then
                return normalGodId
            end
        end
    end
    return enemyGodId
end

function IsGodStolen(godId)
    if not IsGodStealable(godId) then
        return false
    end

    local varName = "戒指_" .. godId .. "_状态"
    return VariableGet(varName) == 1
end

function IsGodStealable(godId)
    Log("IsGodStealable")
    for _, stealableGod in ipairs(canUsedGodList) do
        if stealableGod == godId then
            return true
        end
    end
    return false
end

function SetGodStolen(godId)
    if not IsGodStealable(godId) then
        return
    end
    local varName = "戒指_" .. godId .. "_状态"
    VariableSet(varName, 1)  -- 1表示已被夺取
end

function SetGodAvailable(godId)
    if not IsGodStealable(godId) then
        return
    end
    Log("5-2-1")
    local varName = "戒指_" .. godId .. "_状态"
    VariableSet(varName, 0)  -- 0表示可被夺取
    Log("5-2-2")
end
