function setGodFromUnequip()
    -- 获取并反序列化表
    local currentNotEquipGodList = StringToTable(VariableGet("notEquipGodList"))

    -- 检查是否有至少一个未装备且未被偷取的戒指
    local canStealCount = 0
    for _, godId in ipairs(currentNotEquipGodList) do
        if not IsGodStolen(godId) then
            canStealCount = canStealCount + 1
        end
    end

    -- 只有在至少能偷取一个戒指时才播放动画
    if canStealCount > 0 then
        CursorSetPos_FromPid(g_pid_veyre)
        local x = UnitGetX(g_pid_veyre)
        local z = UnitGetZ(g_pid_veyre)
        WaitTime(0.5)
        EffectPlay("邪竜紋_発動", x, z)
        WaitTime(1.5)
    end

    -- 执行偷取逻辑
    for _, godId in ipairs(currentNotEquipGodList) do
        if not IsGodStolen(godId) then
            StealGodRing(godId, g_pid_lueur)
        end
    end
end
function StealGodRing(godId,fromPid)
    local nearestEnemy, targetX, targetZ = FindNearestEnemyUnitHasNoGod(fromPid)

    if nearestEnemy ~= nil then
        -- 显示效果
        CursorSetPos(targetX, targetZ)
        MapCameraWait()
        EffectPlay("ワープイン_闇", targetX, targetZ)
        UnitShine(nearestEnemy, 0.3)
        WaitTime(0.3)

        -- 为敌方单位装备敌方版本的戒指
        local newGodId = ConvertToEnemyGodId(godId)
        UnitCreateGodUnit(nearestEnemy, newGodId)
        UnitSetHpStock(nearestEnemy,1)

        -- 标记该戒指已被夺取
        SetGodStolen(godId)

        --CobaltNotice("敌方单位在位置(" .. targetX .. "," .. targetZ .. ")装备了戒指: " .. newGodId)
    end
end

function StealGodRingFromPlayer()
    local transferableEnemies = GetTransferableEnemyCount()
    if transferableEnemies <= 0 then
        return  -- 没有可装备戒指的敌人，不进行偷取
    end

    -- 安全地获取表数据
    local currentPlayerGodTable = StringToTable(VariableGet("player_God_Table"))
    local currentNotEquipGodList = StringToTable(VariableGet("notEquipGodList"))

    -- 获取ヴェイル的位置
    local veyleX, veyleZ = 0, 0
    if UnitExistOnMap("PID_M017_ヴェイル") then
        veyleX = UnitGetX("PID_M017_ヴェイル")
        veyleZ = UnitGetZ("PID_M017_ヴェイル")
    else
        -- 如果ヴェイル不存在，则直接返回
        return
    end

    local stealablePlayerRings = {}

    for _, equippedInfo in ipairs(currentPlayerGodTable) do
        -- 确保数据格式正确
        if type(equippedInfo) == "table" and #equippedInfo >= 2 then
            local unitPid = equippedInfo[1]  -- PID在索引1
            local godId = equippedInfo[2]    -- GID在索引2

            -- 检查该戒指是否可偷取且未被偷取过
            if IsGodStealable(godId) and not IsGodStolen(godId) then
                -- 检查单位是否处于结合状态
                local unit = UnitGetByPID(unitPid)
                local isEngaging = false
                if unit ~= nil then
                    isEngaging = UnitGetEngaging(unit)
                end

                -- 只有非结合状态的单位才会被偷取
                if not isEngaging then
                    if UnitExistOnMap(unitPid) then
                        local unitX = UnitGetX(unitPid)
                        local unitZ = UnitGetZ(unitPid)
                        local engageCount = UnitGetEngageCount(unitPid)

                        -- 计算与ヴェイル的距离
                        local distance = 二点間距離(veyleX, veyleZ, unitX, unitZ)

                        table.insert(stealablePlayerRings, {
                            pid = unitPid,
                            ring = godId,
                            x = unitX,
                            z = unitZ,
                            engageCount = engageCount,
                            distance = distance  -- 添加距离信息
                        })
                    end
                end
            end
        end
    end

    -- 按距离排序，距离越近优先级越高
    table.sort(stealablePlayerRings, function(a, b)
        return a.distance < b.distance
    end)

    local stealCount = math.min(2, #stealablePlayerRings, transferableEnemies)

    -- 如果没有可偷取的戒指，直接返回
    if stealCount <= 0 then
        return
    end

    CursorSetPos_FromPid(g_pid_veyre)
    local x = UnitGetX(g_pid_veyre)
    local z = UnitGetZ(g_pid_veyre)
    WaitTime(0.5)
    EffectPlay("邪竜紋_発動", x, z)
    WaitTime(1.5)

    local stolenRings = {}
    for i = 1, stealCount do
        table.insert(stolenRings, stealablePlayerRings[i])
    end

    for _, ringInfo in ipairs(stolenRings) do
        local godId = ringInfo.ring
        local unitPid = ringInfo.pid
        local unitX = ringInfo.x
        local unitZ = ringInfo.z
        local engageCount = ringInfo.engageCount  -- 获取保存的结合气层数

        local engageCountVar = "戒指_" .. godId .. "_计数"
        VariableSet(engageCountVar, engageCount)

        -- 找到最近的未装备戒指的敌方单位
        local nearestEnemy,targetX,targetZ = FindNearestEnemyUnitHasNoGod(unitPid)

        if nearestEnemy ~= nil then
            -- 显示效果
            CursorSetPos(unitX, unitZ)
            MapCameraWait()
            EffectPlay("ワープアウト", unitX, unitZ)
            UnitShine(UnitGetByPID(unitPid), 0.3)
            WaitTime(1.0)

            -- 从玩家单位移除戒指
            UnitSetGodUnit(unitPid, nil)

            CursorSetPos(targetX, targetZ)
            MapCameraWait()
            EffectPlay("ワープイン_闇", targetX, targetZ)
            UnitShine(UnitGetByPID(nearestEnemy), 0.3)
            WaitTime(1.0)

            -- 为敌方单位装备敌方版本的戒指
            local enemyGodId = ConvertToEnemyGodId(godId)
            UnitCreateGodUnit(nearestEnemy, enemyGodId)

            -- 标记该戒指已被偷取
            SetGodStolen(godId)

            --CobaltNotice("从玩家单位 " .. unitPid .. " 处偷取了戒指 " .. godId .. " 并装备给敌方单位")
            --CobaltNotice("结合数 " .. VariableGet(engageCountVar))
        end
    end

    CheckPlayerHasRings()
end



-- 检查被击杀敌方单位的戒指
function checkDeadEnemy()
    local deadUnit = MindGetEventUnit()  -- 获取被击杀的单位
    local deadX = UnitGetX(deadUnit)
    local deadZ = UnitGetZ(deadUnit)
    if deadUnit == nil then
        return
    end

    local enemyGodId = UnitGetGodUnit(deadUnit)  -- 获取敌方单位装备的戒指
    if enemyGodId == nil then
        return  -- 没有装备戒指，直接返回
    end

    -- 将敌方戒指ID转换为我方版本
    local playerGodId = ConvertToNormalGodId(enemyGodId)

    if playerGodId == enemyGodId then
        return  -- 转换失败，不是我们需要处理的戒指
    end

    -- 安全地获取表数据
    local currentNotEquipGodList = StringToTable(VariableGet("notEquipGodList"))
    local currentPlayerGodTable = StringToTable(VariableGet("player_God_Table"))

    -- 检查该戒指是否在notEquipGodList中
    local isInNotEquipList = false
    for _, godId in ipairs(currentNotEquipGodList) do
        if godId == playerGodId then
            isInNotEquipList = true
            break
        end
    end
    if not isInNotEquipList then
        -- 不在notEquipGodList中，搜寻player_God_Table找到原先装备该戒指的单位
        for _, equippedInfo in ipairs(currentPlayerGodTable) do

            if type(equippedInfo) == "table" and #equippedInfo >= 2 and equippedInfo[2] == playerGodId then
                -- 找到原先装备该戒指的单位，重新为其装备戒指
                UnitSetGodUnit(equippedInfo[1], playerGodId)  -- 使用索引1的PID

                local engageCountVar = "戒指_" .. playerGodId .. "_计数"
                local engageCount = VariableGet(engageCountVar)
                --CobaltNotice("戒指_" .. engageCount .. "_计数")
                local unit = UnitGetByPID(equippedInfo[1])
                if unit ~= nil then
                    UnitSetEngageCount(unit, engageCount)
                end
                local targetX = UnitGetX(unit)
                local targetZ = UnitGetZ(unit)

                -- 创建死亡敌人信息对象
                g_deadEnemyList = {deadX,deadZ,targetX,targetZ}

                SetGodAvailable(playerGodId)
                break
            end
        end
    else
        -- 在notEquipGodList中，只刷新其偷取状态
        --CobaltNotice(playerGodId)
        SetGodAvailable(playerGodId)
    end
end

function ConvertToEnemyGodId(godId)
    if not IsGodStealable(godId) then
        return godId
    end

    if string.sub(godId, 1, 4) == "GID_" then
        local baseName = string.sub(godId, 5)
        if baseName then
            return "GID_M017_敵" .. baseName
        end
    end
    return godId
end

function ConvertToNormalGodId(enemyGodId)
    Log(enemyGodId)
    if string.sub(enemyGodId, 1, 10) == "GID_M017_敵" then
        local baseName = string.sub(enemyGodId, 11)
        Log(baseName)
        if baseName then
            local normalGodId = "GID_" .. baseName
            Log(normalGodId)
            -- 检查转换后的戒指是否在可夺取列表中
            if IsGodStealable(normalGodId) then
                return normalGodId
            end
        end
    end
    return enemyGodId
end

function IsGodStolen(godId)
    if not IsGodStealable(godId) then
        return false
    end

    local varName = "戒指_" .. godId .. "_状态"
    return VariableGet(varName) == 1
end

function IsGodStealable(godId)
    Log("IsGodStealable")
    for _, stealableGod in ipairs(canUsedGodList) do
        if stealableGod == godId then
            return true
        end
    end
    return false
end

function SetGodStolen(godId)
    if not IsGodStealable(godId) then
        return
    end
    local varName = "戒指_" .. godId .. "_状态"
    VariableSet(varName, 1)  -- 1表示已被夺取
end

function SetGodAvailable(godId)
    if not IsGodStealable(godId) then
        return
    end
    local varName = "戒指_" .. godId .. "_状态"
    VariableSet(varName, 0)  -- 0表示可被夺取
end

function GetTransferableEnemyCount()
    local count = 0
    local index = ForceUnitGetFirst(FORCE_ENEMY)

    while index ~= nil do
        local godUnit = UnitGetGodUnit(index)
        local jobId = UnitGetJID(index)
        if godUnit == nil and jobId ~= "JID_異形竜" then
            count = count + 1
        end
        index = ForceUnitGetNext(index)
    end

    return count
end

function CheckPlayerHasRings()
    -- 检查所有我方单位是否都没有戒指
    local hasRing = false
    local index = ForceUnitGetFirst(FORCE_PLAYER)

    while index ~= nil do
        local godUnit = UnitGetGodUnit(index)
        if godUnit ~= nil then
            hasRing = true
            break
        end
        index = ForceUnitGetNext(index)
    end

    -- 如果我方没有任何单位拥有戒指，则判定失败
    if not hasRing then
        VariableSet("敗北", 1)
    end
end
