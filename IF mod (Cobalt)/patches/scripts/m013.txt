function TerraSet()
	TerrainSetBegin()
	Log("TerrainSetBegin")
	for x = 1, g_Width do
		for z = 1, g_Height do 
			if TerrainGet(x, z) == "TID_砂漠" then
				TerrainSet(x, z, "TID_平地")
			end
			
		end
	end
	Log("TerrainSetEnd")
	TerrainSetEnd()
end

function 飛行ユニットの０ダメージ攻撃を許可()
	if DifficultyGet() == DIFFICULTY_LUNATIC then
		
		local index	= ForceUnitGetFirst( FORCE_ENEMY )
		
		while ( index != nil ) do
			
			if (UnitGetPID( index ) == "PID_M012_IFMOD_ランスファイター" ) then
				AiSetRejectPower0Attack( index, false )
			end
			
			index = ForceUnitGetNext( index )
		end
	end
end

function Active_Reinforcement_AI()
	Log("Active_Reinforcement_AI")
	local index	= ForceUnitGetFirst( FORCE_ENEMY )
	while ( index != nil ) do
		local pid = UnitGetPID( index ) 
		if pid == "PID_M013_マージナイト" or pid == "PID_M013_アーチャー" or pid == "PID_M013_マージ" or pid == "PID_M013_上忍" or pid == "PID_M013_アクスファイター" or pid == "PID_M013_セイジ" or pid == "PID_M013_忍" or pid == "PID_M013_ソードマスター" or pid == "PID_M013_ダークマージ" or pid == "PID_M013_ベルセルク" or pid == "PID_M013_ソードファイター" or pid == "PID_M013_スナイパー" or pid == "PID_M013_金鵄武者" then 
			AiSetActiveAll(FORCE_ENEMY,pid, true)
		end
		index = ForceUnitGetNext( index )
	end	
end

function 離脱イベント()
	local unit = MindGetUnit()
	if not UnitExistOnMap( unit ) then
		return
	end
	
	UnitRotation( g_pid_lueur, ROTATE_UP)
	UnitMoveWait()
	
	VariableSet( "勝利", 1 )
end

function MapEnding()
	
	Log("MapEnding");
	Log(VariableGet(g_key_visit_1))
	Log(VariableGet(g_key_visit_2))
	Log(VariableGet(g_key_visit_3))
	Log(VariableGet(g_key_visit_4))
	Log(VariableGet(g_key_visit_5))
	Log(VariableGet(g_key_visit_6))
	local village_Visited = VariableGet(g_key_visit_1) + VariableGet(g_key_visit_2) + VariableGet(g_key_visit_3) + VariableGet(g_key_visit_4) + VariableGet(g_key_visit_5) + VariableGet(g_key_visit_6) 
	Log(village_Visited)
	if village_Visited > 3 then
		Log(g_destroy)
		Dialog("MID_TUT_M013_REWARD1")
		ItemGain(nil, "IID_10000G")
		ItemGain(nil, "IID_ブーツ")
		return
	end

	if village_Visited > 2 then
		Log(g_destroy)
		Dialog("MID_TUT_M013_REWARD2")
		ItemGain(nil, "IID_10000G")
		return
	end

	if village_Visited < 3 then
		Log(g_destroy)
		Dialog("MID_TUT_M013_REWARD3")
		ItemGain(nil, "IID_5000G")
		return
	end
end
