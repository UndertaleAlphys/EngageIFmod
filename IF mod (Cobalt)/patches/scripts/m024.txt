function Startup()
	
	Log("Startup")
	WinRuleSetEnemyNumberLessThanOrEqualTo( -1 )	-- ○人以下になったらゲームクリア。0で敵全滅。-1を設定したら判定無効化

    WinRuleSetMID( "MID_RULE_M024_WIN", 15 )
	
	フラグ登録()
	イベント登録()
	
end

function フラグ登録()

end

function イベント登録()

	EventEntryTurn(勝利条件_M024, 1, 1, FORCE_PLAYER)
    EventEntryTurn(ClearLeuerStatus, 1, 1, FORCE_PLAYER)
	-- --------------------------------------------------
	EventEntryEscape(離脱イベント, 6, 13, "PID_リュール", condition_true)
	
end

	Log("Cleanup")

function MapOpening()
	
	Log("MapOpening")
    InitMapM024()
	UnitSetStatus("PID_リュール",UNIT_STATUS_DEFECT)
    UnitSetGodUnit("PID_リュール",nil)
end

function InitMapM024()
    Log("InitMapM024")
    TerrainSetBegin()
        TerrainSet(13, 10, "TID_建造物")
        TerrainSet(12, 10, "TID_建造物")
        TerrainSet(12, 11, "TID_建造物")
        TerrainSet(24, 25, "TID_建造物")
        TerrainSet(27, 23, "TID_建造物")
        TerrainSet(28, 23, "TID_建造物")
    TerrainSetEnd()
end

function ClearLeuerStatus()
    UnitClearStatus("PID_リュール",UNIT_STATUS_DEFECT)
    UnitClearStatus("PID_リュール",UNIT_STATUS_FIXED)
end

	WinRule()
	CursorAnimeDelete()
end

function 離脱イベント()
	local unit = MindGetUnit()
	if not UnitExistOnMap( unit ) then
		return
	end
	
	UnitRotation( g_pid_lueur, ROTATE_UP)
	UnitMoveWait()
	
	VariableSet( "勝利", 1 )
end
