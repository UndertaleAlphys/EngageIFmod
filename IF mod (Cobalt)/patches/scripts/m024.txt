function ProcessEnemyDie()
    local unit = MindGetEventUnit()
    local pid = UnitGetPID(unit)
    if pid == "PID_IFMOD_M024_上忍_BOSS1" or
       pid == "PID_IFMOD_M024_上忍_BOSS2" or
       pid == "PID_IFMOD_M024_ソードマスター_BOSS1" or
       pid == "PID_IFMOD_M024_ソードマスター_BOSS2" or
       pid == "PID_IFMOD_M024_ハルバーディア_BOSS1" or
       pid == "PID_IFMOD_M024_ハルバーディア_BOSS2" or
       pid == "PID_IFMOD_M024_ウルフナイト_BOSS1" or
       pid == "PID_IFMOD_M024_ウルフナイト_BOSS2" then
       Log("ProcessEnemyDie")
       local DieCount = VariableGet(g_defeated_FormidableEnemy)
       DieCount = DieCount + 1
       VariableSet(g_defeated_FormidableEnemy,DieCount)

       if DieCount == 8 then
           ClearMagicBarrier()
       end
    end
end

function ClearUnmovableStatus1()
    Log("ClearUnmovableStatus1")
    local enemyUnits = ForceUnitGetFirst(FORCE_ENEMY)
    while enemyUnits ~= nil do
        local x = UnitGetX(enemyUnits)
        local z = UnitGetZ(enemyUnits)
        if x >= 1 and x <= 12 and z >= 1 and z <= 8 then
            UnitClearStatus(enemyUnits, UNIT_STATUS_MOVE_NOT_ALLOW)
        end
        enemyUnits = ForceUnitGetNext(enemyUnits)
    end
end

function getItemSecret()
    if VariableGet(g_get_item) == 1 then
        return false
    end
    ItemGain(MindGetUnit(), "IID_巧者の和弓")
    EffectDelete("ドロップアイテム", 19, 30)
    TerrainSetOne(19, 30, "TID_砂漠_M024")

    local unit = UnitGetByPos(19, 30)
    if unit ~= nil then
        EffectPlay("竜化_G1", 19, 30)
        WaitTime(0.5)
        local targetX, targetY = FindNearestValidPosition(19, 25)
        if targetX ~= nil and targetY ~= nil then
            UnitSetPos(unit, targetX, targetY)
            CursorSetPos(targetX, targetY)
            MapCameraWait()
        end
    end
    VariableSet(g_get_item,1)
end

function FindNearestValidPosition(startX, startY)
    local queue = {}
    local visited = {}
    local directions = {{0, 1}, {1, 0}, {0, -1}, {-1, 0}}

    table.insert(queue, {x = startX, y = startY})
    visited[startX .. "," .. startY] = true

    while #queue > 0 do
        local current = table.remove(queue, 1)
        if IsPositionValid(current.x, current.y) then
            return current.x, current.y
        end
        for _, dir in ipairs(directions) do
            local newX = current.x + dir[1]
            local newY = current.y + dir[2]
            local key = newX .. "," .. newY
            if not visited[key] and IsWithinBounds(newX, newY) and IsPassable(newX, newY) then
                table.insert(queue, {x = newX, y = newY})
                visited[key] = true
            end
        end
    end

    return nil, nil
end

function IsPassable(x, y)
    local terrainCost = TerrainGetMoveCost(x, y)
    return terrainCost != "COST_不可" and terrainCost != "COST_空"
end

function IsWithinBounds(x, y)
    return x >= 1 and x <= g_map_width and y >= 1 and y <= g_map_height
end

function IsPositionValid(x, y)
    local unit = UnitGetByPos(x, y)
    if unit ~= nil then
        return false
    end
    return IsPassable(x, y)
end
