function ClearAllMiasma()
	MapOverlapSetBegin()
	for i = 1, g_width do
		for j = 1, g_height do
			if ( MapOverlapGet( i, j ) == "TID_瘴気_永続") then
				MapOverlapSet( i , j, "TID_無し" )
			end
		end
	end
	MapOverlapSetEnd()
end

function 召喚渦作成(areaname)
	CursorSetPos_FromPid(g_pid_boss)
	MapCameraWait()

	local countD = DisposGetGroupCount(areaname)
	countD = countD - 1

		-- グループの人数分処理を回し、ユニットの座標を走査、取得する
		for i = 0, countD do
			local unitX = DisposGetUnitX(areaname, i,DifficultyGet())
			local unitZ = DisposGetUnitZ(areaname, i,DifficultyGet())

			if unitX != -1 then
				EffectCreate( "召喚渦", unitX,unitZ )
			end
		end

end

function 召喚渦消去と配置(areaname)
	CursorSetPos_FromPid(g_pid_boss)
	MapCameraWait()

	Dispos( areaname, DISPOS_FLAG_FOCUS + DISPOS_FLAG_WARP)
	Yield()

	countD = DisposGetGroupCount(areaname)
	countD = countD - 1
	for i = 0, countD do
		local unitX = DisposGetUnitX(areaname, i,DifficultyGet())
		local unitZ = DisposGetUnitZ(areaname, i,DifficultyGet())
		if unitX != -1 then
			EffectDelete( "召喚渦", unitX,unitZ )
		end
	end
	Yield()

end

function Fire_Expansion_PlayerTurn()
    if VariableGet(g_Deadline_Begin) == 1 then
        local turn = MapGetTurn()
        if (turn % 2 == 1) then
             CursorSetPos(4,1)
             MapCameraWait()
             Dispos( "Reinforcement_Back", DISPOS_FLAG_FORCED)
             Yield()
             WaitTime(1.5)
        end
    end

	if VariableGet(g_door3_opened) == 1 then
        SpawnReinforcements()
    end
end

function SpawnReinforcements()

    local roll = VariableGet(g_reinforcement_counter)

    local reinforcement_config = ""

    if roll == 0 then
        reinforcement_config = "Enemy_Reinforcement1"
    elseif roll == 1 then
        reinforcement_config = "Enemy_Reinforcement2"
    elseif roll == 2 then
        reinforcement_config = "Enemy_Reinforcement3"
    elseif roll == 3 then
        reinforcement_config = "Enemy_Reinforcement4"
    end


    if reinforcement_config ~= "" then
        CursorSetPos( 24, 1 )
        WaitTime(1.5)
        Dispos(reinforcement_config, DISPOS_FLAG_FORCED)
        Yield()
        WaitTime(1.5)
    end

    roll = (roll + 1) % 4
    VariableSet(g_reinforcement_counter,roll)
end

function Fire_Expansion_EnemyTurn()
	Fire_Expansion()
end

function Fire_Expansion()
	local directions = { {0, 1}, {1, 0}, {0, -1}, {-1, 0} }
    local toUpdate = {}
    local visited = {}
    for x = 1, g_width do
        for z = 1, g_height do
            if TerrainGet(x, z) == "TID_炎上_M015" then
				 for _, dir in ipairs(directions) do
                    local nx = x + dir[1]
                    local nz = z + dir[2]
                    local key = nx .. "," .. nz
                    if not visited[key] then
                        if IsWithinBounds(nx, nz) then
                            local cost = TerrainGetMoveCost(nx, nz)
                            if cost ~= "COST_不可" then
                                table.insert(toUpdate, {x = nx, z = nz})
                                visited[key] = true
                            end
                        end
                    end
                end
			end
		end
	end
	if #toUpdate > 0 then
        TerrainSetBegin()
		MapOverlapSetBegin()
        for _, pos in ipairs(toUpdate) do
            if TerrainGet(pos.x, pos.z) ~= "TID_炎上_M015" then
                TerrainSet(pos.x, pos.z, "TID_炎上_M015")
            end
			if MapOverlapGet(pos.x, pos.z) ~= "TID_炎上_M015_Overlap" then
                MapOverlapSet(pos.x, pos.z, "TID_炎上_M015_Overlap")
            end
        end
		MapOverlapSetEnd()
        TerrainSetEnd()
    end
end

function DamageEvent()
	for x = 1, g_width do
        for z = 1, g_height do
            if TerrainGet(x, z) == "TID_炎上_M015" then
                local unit = UnitGetByPos(x, z)
                if unit ~= nil and UnitExistOnMap(unit) then
					MapDamageM015(unit)
					UnitDie( unit )
                end
            end
        end
    end
end

function FireDamage(unit)
    MapDamageBegin()
    MapDamageAdd(unit, 100)  -- 固定造成100点伤害
    MapDamageEnd()
end

function IsWithinBounds(x, z)
    return x > 0 and x < g_width and z > 0 and z < g_height
end

	WaitTime(0.5)

		UnitMovePosFromPos(23,1,23,0)
	end

		UnitMovePos(24,1,24,0)
	end

		Talk( "MID_EV7" )

	end
