function ClearAllMiasma()
	MapOverlapSetBegin()
	for i = 1, g_width do
		for j = 1, g_height do
			if ( MapOverlapGet( i, j ) == "TID_瘴気_永続") then
				MapOverlapSet( i , j, "TID_無し" )
			end
		end
	end
	MapOverlapSetEnd()
end

function Fire_Expansion()
	CobaltNotice("Fire_Expansion")
	local directions = { {0, 1}, {1, 0}, {0, -1}, {-1, 0} }
    local toUpdate = {}  
    local visited = {} 
    for x = 1, g_width do
        for z = 1, g_height do
            if TerrainGet(x, z) == "TID_炎上_M015" then
				 for _, dir in ipairs(directions) do
                    local nx = x + dir[1]
                    local nz = z + dir[2]
                    local key = nx .. "," .. nz
                    if not visited[key] then
                        if IsWithinBounds(nx, nz) then
                            local cost = TerrainGetMoveCost(nx, nz)
                            if cost ~= "COST_不可" then
                                table.insert(toUpdate, {x = nx, z = nz})
                                visited[key] = true
                            end
                        end
                    end
                end
			end
		end
	end
	CobaltNotice("BFS Complete")
	if #toUpdate > 0 then
        TerrainSetBegin()
		MapOverlapSetBegin()
        for _, pos in ipairs(toUpdate) do
            if TerrainGet(pos.x, pos.z) ~= "TID_炎上_M015" then
                TerrainSet(pos.x, pos.z, "TID_炎上_M015")
            end
			if MapOverlapGet(pos.x, pos.z) ~= "TID_炎上_M015_Overlap" then
                MapOverlapSet(pos.x, pos.z, "TID_炎上_M015_Overlap")
            end
        end
		MapOverlapSetEnd()
        TerrainSetEnd()
    end
	CobaltNotice("Update Terrain")
end

function DamageEvent()
	CobaltNotice("Damage Event Begin")
	for x = 1, g_width do
        for z = 1, g_height do
            if TerrainGet(x, z) == "TID_炎上_M015" then
                local unit = UnitGetByPos(x, z)
                if unit ~= nil and UnitExistOnMap(unit) then
					MapDamageM015(unit)
					UnitDie( unit )
                end
            end
        end
    end
end

function FireDamage(unit)
    MapDamageBegin()
    MapDamageAdd(unit, 100)  -- 固定造成100点伤害
    MapDamageEnd()
end

function IsWithinBounds(x, z)
    return x > 0 and x < g_width and z > 0 and z < g_height
end

function PersonEscape()

	local unit = MindGetUnit()

	local unit_x = UnitGetX(unit)

	WaitTime(0.5)

	UnitRotation( unit, ROTATE_DOWN)

	WaitTime(0.5)

	if (unix_x == 23) then
		UnitMovePosFromPos(23,1,23,0)
	end

	if (unix_x == 24) then
		UnitMovePos(24,1,24,0)
	end
	
	if (ForceUnitGetCount(FORCE_PLAYER) == 1) then
		Talk( "MID_EV7" )
		VariableSet( "勝利", 1 )
	end

	UnitTransfer(unit,FORCE_ABSENT)

end
