function UpdateBuildingTilesList()
    local visited = {}    -- 记录已访问的坐标，避免重复处理

    for _, obstacle in ipairs(g_obstaclePoint) do
        local startX = obstacle[1]
        local startZ = obstacle[2]

        -- 检查地形状态，如果是建造物则进行BFS扩展
        local terrain = TerrainGet(startX, startZ)
        if (terrain == "TID_建造物" or terrain == "TID_瓦礫" )and not visited[startX .. "," .. startZ] then
            -- 使用BFS查找所有相连的相同地形
            local queue = {{x = startX, z = startZ}}
            visited[startX .. "," .. startZ] = true

            while #queue > 0 do
                local current = table.remove(queue, 1)
                local currentX = current.x
                local currentZ = current.z

                table.insert(g_buildingTiles, {currentX, currentZ})

                local directions = {{1, 0}, {-1, 0}, {0, 1}, {0, -1}}
                for _, dir in ipairs(directions) do
                    local nextX = currentX + dir[1]
                    local nextZ = currentZ + dir[2]
                    local key = nextX .. "," .. nextZ

                    if nextX >= 0 and nextX < g_width and nextZ >= 0 and nextZ < g_height and not visited[key] then
                        if TerrainGet(nextX, nextZ) == "TID_建造物" or TerrainGet(nextX, nextZ) == "TID_瓦礫" then
                            table.insert(queue, {x = nextX, z = nextZ})
                            visited[key] = true
                        end
                    end
                end
            end
        end
    end
end

function AiSetCamillaTurnBefore()
    if VariableGet(g_key_camillaActive) == 1 then
        return
    end

    local index = ForceUnitGetFirst(FORCE_PLAYER)

    while index ~= nil do
        local isIndexInRange = IsUnitInAttackRange(attackRangeCamilla,index)

             VariableSet(g_key_camillaActive , 1)

function CamillaAiSetTurnBefore()
    local attackRangeCamilla = GetUnitActualAttackRange(g_pid_camilla)

    -- 检查卡米拉AI是否已经完全激活
    if VariableGet(g_key_camillaActiveRun) == 1 then
        return
    end

    -- 遍历检查全部我方单位，观察其是否在卡米ラ范围内
    local camillaX = UnitGetX(g_pid_camilla)
    local camillaZ = UnitGetZ(g_pid_camilla)
    local index = ForceUnitGetFirst(FORCE_PLAYER)
    local shouldActivate = false

    while index ~= nil do
        local unitX = UnitGetX(index)
        local unitZ = UnitGetZ(index)

        -- 检查单位是否在卡米ラ攻击范围内
        if 二点間距離(unitX, unitZ, camillaX, camillaZ) <= attackRangeCamilla then
            shouldActivate = true
            break
        end

        index = ForceUnitGetNext(index)
    end

    -- 根据检查结果设置AI模式和标记
    if shouldActivate then
        AiSet(g_pid_camilla, AI_AC_EveryTime)
        VariableSet(g_key_camillaActiveRun, 1)
    else
        AiSet(g_pid_camilla, AI_AC_Null)
    end
end

-- 龙脉使用后的处理
function DragonVeinPostUseHandleG005(veinInfo)
    if veinInfo.type == "bomb" then
        DragonVeinEffectBomb(veinInfo.param)
        DraginVein_Skill_Equip(veinInfo.skillId)
    elseif veinInfo.type == "sky" then
        DragonVeinEffectSky()
        TerrainSetBegin()
            TerrainSet(veinInfo.x, veinInfo.z, "TID_床")
        TerrainSetEnd()
        EffectDelete("魔法陣_G001", veinInfo.x, veinInfo.z)
    elseif veinInfo.type == "ground" then
        DragonVeinEffectGround()
        TerrainSetBegin()
            TerrainSet(veinInfo.x, veinInfo.z, "TID_床")
        TerrainSetEnd()
        EffectDelete("魔法陣_G001", veinInfo.x, veinInfo.z)
    end
end


function DragonVeinEffectBomb(column)
    local enemys = GetEnemyUnitsInColumn(column)
    local x,z = CalculateVisualCenter(enemys)
    if x == 0 or z == 0 then
        z = 13
        x = 10 * column + 5
    end
    CursorSetPos(x,z)
    WaitTime(1.0)
    EffectPlay( "カムイ竜脈", x, z )
    EffectWait()

    for _, unit in ipairs(enemys) do
        MapDamageBegin()
            local currentHp = UnitGetHp(unit)
            local map_damage = 10;
            local damage = map_damage
            if damage >= currentHp then
                damage = currentHp - 1
            end
            MapDamageAdd(unit, damage)
        MapDamageEnd()
    end

    DestroyObstaclesInColumn(column)
end

function GetUnitCountByColumn()
    leftCount = 0
    middleCount = 0
    rightCount = 0

    leftUnits = {}
    middleUnits = {}
    rightUnits = {}

    -- 遍历所有我方单位
    local index = ForceUnitGetFirst(FORCE_PLAYER)
    while index ~= nil do
        local x = UnitGetX(index)
        local z = UnitGetZ(index)

        -- 根据x坐标判断单位所在的列区域
        if x >= g_map_left_column_low and x <= g_map_left_column_high then
            leftCount = leftCount + 1
            table.insert(leftUnits, index)
        elseif x >= g_map_middle_column_low and x <= g_map_middle_column_high then
            middleCount = middleCount + 1
            table.insert(middleUnits, index)
        elseif x >= g_map_right_column_low and x <= g_map_right_column_high then
            rightCount = rightCount + 1
            table.insert(rightUnits, index)
        end

        index = ForceUnitGetNext(index)
    end

    local targetColumn = GetMostPopulatedAvailableColumn()
    -- 确定人数最多的列，并计算视觉中心点
    if targetColumn == 0 then
        -- 计算左侧列的视觉中心点
        local centerX, centerZ = CalculateVisualCenter(leftUnits)
        local enemyUnits = GetEnemyUnitsInColumn(0)
        return 0, leftUnits, centerX, centerZ, enemyUnits
    elseif targetColumn == 1 then
        -- 计算中间列的视觉中心点
        local centerX, centerZ = CalculateVisualCenter(middleUnits)
        local enemyUnits = GetEnemyUnitsInColumn(1)
        return 1, middleUnits, centerX, centerZ, enemyUnits
    else
        -- 计算右侧列的视觉中心点
        local centerX, centerZ = CalculateVisualCenter(rightUnits)
        local enemyUnits = GetEnemyUnitsInColumn(2)
        return 2, rightUnits, centerX, centerZ, enemyUnits
    end
end

function locationCanUsed(x,z)
    local unit = UnitGetByPos(x, z)
    if unit == nil then
        return true
    end

    if UnitGetPID(unit) == g_pid_camilla then
        return true
    end
    return false
end

-- 判断哪一列人数最多且可以使用
function GetMostPopulatedAvailableColumn()
    local canLeftUse = locationCanUsed(2, 25)
    local canMiddleUse = locationCanUsed(3, 25)
    local canRightUse = locationCanUsed(4, 25)
    -- 检查各列是否可用并比较人数
    local maxCount = -1
    local targetColumn = -1

    -- 检查左侧列
    if canLeftUse and leftCount > maxCount then
        maxCount = leftCount
        targetColumn = 0
    end
    -- 检查中间列
    if canMiddleUse and middleCount > maxCount then
        maxCount = middleCount
        targetColumn = 1
    end
    -- 检查右侧列
    if canRightUse and rightCount > maxCount then
        maxCount = rightCount
        targetColumn = 2
    end
    return targetColumn
end


-- 计算单位列表的视觉中心点
function CalculateVisualCenter(units)
    if #units == 0 then
        return 0, 0
    end

    local totalX = 0
    local totalZ = 0

    for _, unit in ipairs(units) do
        totalX = totalX + UnitGetX(unit)
        totalZ = totalZ + UnitGetZ(unit)
    end

    -- 返回平均坐标作为视觉中心点
    return totalX / #units, totalZ / #units
end

-- 获取指定列的敌方单位列表
function GetEnemyUnitsInColumn(column)
    local enemyUnits = {}

    -- 确定列的x坐标范围
    local low, high
    if column == 0 then
        low = g_map_left_column_low
        high = g_map_left_column_high
    elseif column == 1 then
        low = g_map_middle_column_low
        high = g_map_middle_column_high
    else
        low = g_map_right_column_low
        high = g_map_right_column_high
    end

    -- 遍历所有敌方单位
    local index = ForceUnitGetFirst(FORCE_ENEMY)
    while index ~= nil do
        local x = UnitGetX(index)

        -- 检查敌方单位是否在指定列中
        if x >= low and x <= high then
            table.insert(enemyUnits, index)
        end

        index = ForceUnitGetNext(index)
    end

    return enemyUnits
end

function DragonVeinUsedByEmenyEffect(column,unitListPlayer,unitListEnemy)
    for _, unit in ipairs(unitListPlayer) do
        MapDamageBegin()
            local currentHp = UnitGetHp(unit)
            local damage = 10;
            if damage >= currentHp then
                damage = currentHp - 1
            end
            MapDamageAdd(unit, damage)
        MapDamageEnd()
    end

    WaitTime(1.0)

    CursorSetPos(10*column+5,13)
    MapCameraWait()

    for _, unit in ipairs(unitListEnemy) do
        local indexX = UnitGetX(unit)
        local indexZ = UnitGetZ(unit)
        --EffectPlay( "水晶修復" , indexX, indexZ )
        UnitSetPrivateSkill(unit, "SID_天駆")
    end

    DestroyObstaclesInColumn(column)
end

function DestroyObstaclesInColumn(column)
    -- 确定列的x坐标范围
    local low, high
    if column == 0 then
        low = g_map_left_column_low
        high = g_map_left_column_high
    elseif column == 1 then
        low = g_map_middle_column_low
        high = g_map_middle_column_high
    else
        low = g_map_right_column_low
        high = g_map_right_column_high
    end

    -- 遍历G005_Gimmick中定义的障碍物列表
    for _, obstacle in ipairs(g_obstaclePoint) do
        local x = obstacle[1]  -- 障碍物的x坐标
        local z = obstacle[2]  -- 障碍物的z坐标
        local ex = obstacle[3] -- 障碍物の演出x坐标
        local ez = obstacle[4] -- 障碍物の演出z坐标

        -- 检查障碍物是否在指定列范围内
        if x >= low and x <= high then
            -- 检查障碍物是否仍然存在（TID_建造物表示未被破坏的障碍物）
            if TerrainGet(x, z) == "TID_建造物" or TerrainGet(x, z) == "TID_瓦礫" then
                -- 破坏障碍物
                DestroyObstacle(x, z, ex, ez)
                WaitTime(0.2)
            end
        end
    end
end

function DestroyObstacle(x, z, ex, ez)
    -- 检查是否为需要两次破坏的障碍物
    local isTwiceBreakable = false
    for _, obstacle in ipairs(g_obstaclePointTwice) do
        if obstacle[1] == x and obstacle[2] == z then
            isTwiceBreakable = true
            break
        end
    end

    -- 检查是否为一次就能完全破坏的障碍物（在特定条件下）
    local isAtOnceBreakable = false
    for _, obstacle in ipairs(g_obstaclePointAtOnce) do
        if obstacle[1] == x and obstacle[2] == z then
            isAtOnceBreakable = true
            break
        end
    end

    -- 检查是否为特殊障碍物
    local isSpecial = (x == 21 and z == 23) or (x == 22 and z == 20) or
                      (x == 11 and z == 18) or (x == 9 and z == 18)

    -- 根据不同类型采用不同的破坏方式
    if isSpecial then
        障害物破壊_特殊(x, z, "TID_床")
    else
        -- 检查当前地形状态决定破坏方式
        local currentTerrain = TerrainGet(x, z)

        if currentTerrain == "TID_建造物" then
            if isAtOnceBreakable == true then
                破壊処理(x,z,true)
            else
                破壊処理(x,z,false)
            end
        elseif currentTerrain == "TID_瓦礫" then
            破壊処理(x,z,true)
        end
    end
end

		god = UnitGetGodUnit( MindGetUnit() )

end

function condition_カミラが遠距離攻撃された()

	if VariableGet( g_key_camillaAttackedFar ) == 1 then
		return false
	end

	if モードはルナティック() == false then
		return false
	end

	if AiGetActive( g_pid_boss ) == true then
		return false
	end

	local unit = MindGetUnit()
	local x = UnitGetX( unit )
	local z = UnitGetZ( unit )
	local camillaX = UnitGetX( g_pid_boss )
	local camillaZ = UnitGetZ( g_pid_boss )

	return ( 二点間距離( x, z, camillaX, camillaZ ) > 8 )

end

function condition_増援_カミラ非遠距離攻撃後()

	if VariableGet( g_key_reinforcementCamillaAttackedFar ) == 0 then

		if モードはルナティック() == false then
			return false
		end

	elseif VariableGet( g_key_reinforcementCamillaAttackedFar ) == 1 then

		return ( VariableGet( g_key_camillaActiveRun ) == 0 )

	end
end

	end

end
