function 離脱イベント()

	if not UnitExistOnMap( unit ) then
		return
	end
	
	UnitRotation( g_pid_lueur, ROTATE_UP)
	UnitMoveWait()
	
	VariableSet( "勝利", 1 )
end

function Reinforcement()
    local Turn = MapGetTurn()

    if Turn < 8 then
        return
    end

    Turn = Turn % 4

    if Turn == 1 then
        Reinforcement1()
    end

    if Turn == 2 then
        Reinforcement2()
    end

    if Turn == 3 then
        Reinforcement3()
    end

    if Turn == 0 then
        Reinforcement4()
    end
end

function InitMapM026()
	Log("InitMapM026")
    TerrainSetBegin()
    MapOverlapSetBegin()
	for x = 1, g_map_width do
		for z = 1, g_map_height do
			if MapOverlapGet(x, z) == "TID_霧_永続" then
				MapOverlapSet(x, z, "TID_無し")
			end
            if TerrainGet(x, z) == "TID_弓砲台" then
                TerrainSet(x, z, "TID_弓砲台_暗器砲台")
            end
		end
	end

    TerrainSet(29,13,"TID_王座")

	for _, point in ipairs(fogPoint) do
		MapOverlapSet(point[1], point[2], "TID_霧_永続")
	end
	MapOverlapSetEnd()
    TerrainSetEnd()

	local magicCirclePoints = {}
    for _, config in ipairs(g_dragonVeinConfig_M026) do
        table.insert(magicCirclePoints, config.point)
    end
    DragonVeinCreateMagicCircle(magicCirclePoints)

    local terrainPoints = {}
    for _, config in ipairs(g_dragonVeinConfig_M026) do
        table.insert(terrainPoints, {config.point[1], config.point[2], "TID_龙脉_M026_Block"})
    end
    DragonVeinSetTerrain(terrainPoints)
end

function DragonVeinUsed()
    local unit = MindGetUnit()
    local x = UnitGetX(unit)
    local z = UnitGetZ(unit)
    local dragonVeinPoints = {}
    for _, config in ipairs(g_dragonVeinConfig_M026) do
        table.insert(dragonVeinPoints, {
            config.point,
            config.skillId,
            config.type,
            config.param
        })
    end


    local veinInfo = DragonVeinPreUseCheck(
        dragonVeinPoints,
        x,
        z
    )

    if veinInfo then
        DragonVeinPostUseHandleM026(veinInfo)
    end
end

function DragonVeinPostUseHandleM026(veinInfo)
	Log("DragonVeinPostUseHandleM026")
    TerrainSetBegin()
        TerrainSet(veinInfo.x, veinInfo.z, "TID_道")
    TerrainSetEnd()

    EffectDelete("魔法陣_G001", veinInfo.x, veinInfo.z)
    CursorSetPos(veinInfo.x, veinInfo.z)
    WaitTime(1.0)
    EffectPlay("水球_発動", veinInfo.x, veinInfo.z)
    WaitTime(1.5)

    for _, config in ipairs(g_dragonVeinConfig_M026) do
        if veinInfo.x == config.point[1] and veinInfo.z == config.point[2] then
            local distOrg = CursorGetDistanceMode()

            CursorSetDistanceMode(CURSOR_DISTANCE_MIDDLE)
            MapCameraWait()

            CursorSetPos(26, 13)
            WaitTime(1.0)
            EffectPlay("水球", 26, 13)
            WaitTime(1.5)

            CursorSetDistanceMode(distOrg)
            MapCameraWait()

            DragonVeinApplyEffect()

            break
        end
    end
end

end

function DragonVeinUsedByEnemyPredict()
	Log("DragonVeinUsedByEnemyPredict")
	if MapGetTurn() == 1 then
		return
	end
	local Turn = MapGetTurn() % 2
	local bossX = UnitGetX(g_pid_boss)
	local bossZ = UnitGetZ(g_pid_boss)

	CursorSetPos(bossX, bossZ)	
	MapCameraWait()

	EffectCreate("邪竜紋_準備",bossX, bossZ)
	WaitTime(1.0)

	MapDangerRangeUpdate()
end

function DragonVeinUsedByEnemyProcess()
	Log("DragonVeinUsedByEnemyProcess")
	if MapGetTurn() == 1 then
		return
	end
	
	local bossX = UnitGetX(g_pid_boss)
	local bossZ = UnitGetZ(g_pid_boss)

	CursorSetPos(bossX, bossZ)	
	MapCameraWait()

	EffectDelete("邪竜紋_準備", bossX, bossZ)
	EffectPlay("邪竜紋_発動", bossX, bossZ)
    WaitTime(1.0)

	if MapGetTurn() % 2 == 1 then
		PlayAllAvalancheEffects()
        ProcessDragonVeinDamage(attackRowOddLow, attackRowOddHigh)
		ApplyDragonVeinDamage()
    else
		PlayAllAvalancheEffects()
        ProcessDragonVeinDamage(attackRowEvenLow1, attackRowEvenHigh1)
        ProcessDragonVeinDamage(attackRowEvenLow2, attackRowEvenHigh2)
		ApplyDragonVeinDamage()
    end

	MapRangeClear()
end

function PlayAvalancheEffect(lowRow, highRow)
    EffectPlay("雪崩", g_map_width - 1, (lowRow + highRow) / 2)
end

function PlayAllAvalancheEffects()
	local distOrg = CursorGetDistanceMode()

	CursorSetDistanceMode(CURSOR_DISTANCE_FAR)
	MapCameraWait()

	CursorSetPos(15,13)
	MapCameraWait()
    if MapGetTurn() % 2 == 1 then
        PlayAvalancheEffect(attackRowOddLow, attackRowOddHigh)
    else
        PlayAvalancheEffect(attackRowEvenLow1, attackRowEvenHigh1)
        PlayAvalancheEffect(attackRowEvenLow2, attackRowEvenHigh2)
    end
    WaitTime(4.0)
    MapCameraWait()

	CursorSetDistanceMode(distOrg)
	MapCameraWait()
end

function ProcessDragonVeinDamage(lowRow, highRow)
    Log("ProcessDragonVeinDamage " .. lowRow .. " to " .. highRow)
    for z = lowRow, highRow do
        local blockX = nil
        for x = 1, g_map_width do
            if MapOverlapGet(x, z) == "TID_ブロック_M026" then
                blockX = x
				table.insert(g_dragonVeinBlockList, {
                    x = x,
                    z = z
                })
                break
            end
        end
        for x = 1, g_map_width do
            local unit = UnitGetByPos(x, z)
            if unit and UnitGetForce(unit) == FORCE_PLAYER then
                if blockX == nil or x > blockX then
                    table.insert(g_dragonVeinDamageList, {
                        unit = unit,
                        x = x,
                        z = z
                    })
                end
            end
        end
    end
end

function ApplyDragonVeinDamage()
    if #g_dragonVeinDamageList > 0 or #g_dragonVeinBlockList > 0 then
		if #g_dragonVeinBlockList > 0 then
            MapOverlapSetBegin()
            for _, blockInfo in ipairs(g_dragonVeinBlockList) do
                MapOverlapSet(blockInfo.x, blockInfo.z, "TID_霧_永続")
            end
            MapOverlapSetEnd()
        end

        if #g_dragonVeinDamageList > 0 then
            local centerX = 0
            local centerZ = 0
            for _, damageInfo in ipairs(g_dragonVeinDamageList) do
                centerX = centerX + damageInfo.x
                centerZ = centerZ + damageInfo.z
            end
            centerX = math.floor(centerX / #g_dragonVeinDamageList)
            centerZ = math.floor(centerZ / #g_dragonVeinDamageList)
            
            CursorSetPos(centerX, centerZ)
            MapCameraWait()
            
            MapDamageBegin()
            for _, damageInfo in ipairs(g_dragonVeinDamageList) do
				local damage = UnitGetCapability(damageInfo.unit,CAPABILITY_HP,true)/2
                MapDamageAdd(damageInfo.unit, damage)
                local isEngaging = UnitGetEngaging(damageInfo.unit)

                if isEngaging then
                    local engageTurn = UnitGetEngageTurn(damageInfo.unit)
                    local postEngageTurn = math.max(0,engageTurn+2)
                    UnitSetEngageTurn(damageInfo.unit,postEngageTurn)
                else
                    local preEngageCount = UnitGetEngageMeter(damageInfo.unit)
                    UnitSetEngageMeter(damageInfo.unit,math.max(0,preEngageCount-4))
                end
            end
            MapDamageEnd()
        end
        
        g_dragonVeinDamageList = {}
		g_dragonVeinBlockList = {}
    end
end

function MapDangerRangeUpdate()
    Log("MapDangerRangeUpdate")
    MapRangeClear()
    if MapGetTurn() == 1 then
        return
    end
    
    if MapGetTurn() % 2 == 1 then
		MapRangeAddBegin()
        AddDangerRange(attackRowOddLow, attackRowOddHigh)
		MapRangeAddEnd()
    else
		MapRangeAddBegin()
        AddDangerRange(attackRowEvenLow1, attackRowEvenHigh1)
        AddDangerRange(attackRowEvenLow2, attackRowEvenHigh2)
		MapRangeAddEnd()
    end
end

function AddDangerRange(lowRow, highRow)
    Log("AddDangerRange " .. lowRow .. " to " .. highRow)
    
    for z = lowRow, highRow do
        local blockX = nil
        
        for x = 1, g_map_width do
            if MapOverlapGet(x, z) == "TID_ブロック_M026" then
                blockX = x
                break
            end
        end
        
        for x = 1, g_map_width do
            if blockX == nil or x > blockX then
                MapRangeAdd(x, z)
            end
        end
    end
end
