function DragonVeinUsed()
    local unit = MindGetUnit();
    local x = UnitGetX(unit)
    local z = UnitGetZ(unit)
    local used = ""
    local DragonVeinType = ""
    if(x == 27 and z == 12) then
        used = DraginVein_Skill_Remove("SID_DragonVein_M016_TIDE_0","SID_DragonVein_Used")
        DragonVeinType = "tide"
    end
    if(x == 14 and z == 13) then
        used = DraginVein_Skill_Remove("SID_DragonVein_M016_TIDE_1","SID_DragonVein_Used")
        DragonVeinType = "tide"
    end
    if(x == 19 and z == 3) then
        used = DraginVein_Skill_Remove("SID_DragonVein_M016_TIDE_2","SID_DragonVein_Used")
        DragonVeinType = "tide"
    end
    if(x == 9 and z == 17) then
        used = DraginVein_Skill_Remove("SID_DragonVein_M016_TIDE_3","SID_DragonVein_Used")
        DragonVeinType = "tide"
    end
    if(x == 3 and z == 3) then
        used = DraginVein_Skill_Remove("SID_DragonVein_M016_TIDE_4","SID_DragonVein_Used")
        DragonVeinType = "tide"
        if used == true then
            CursorSetPos(3,3)
            WaitTime( 0.5 )
            EffectPlay( "水球_発動", 3, 3 )
            EffectWait()
            CursorSetPos( 15, 11 )
            MapCameraWait()
            EffectPlay( "水球", 15, 11 )
            EffectWait()
            EffectDelete("魔法陣_G001", 3, 3)
            TerrainSetBegin()
                TerrainSet( 3, 3, "TID_砂浜")
            TerrainSetEnd()
            HighTide()
            return
        end
    end

    if used == true then
        CursorSetPos( 15, 11 )
        WaitTime( 0.5 )
        EffectPlay( "カムイ竜脈", 15, 11 )
        WaitTime( 3.0 )
        EffectDelete("魔法陣_G001", x, z)
        TerrainSetBegin()
            TerrainSet( x, z, "TID_砂浜")
        TerrainSetEnd()
        DragonVeinEffect()
    end
    Log("DragonVeinUsedComplete")
end

function DragonVeinEffect()
    EbbTide()
end

function condition_HighTide()

	local turn = VariableGet( g_key_highTide_turn )

	if turn == 1 then
		VariableSet( g_key_highTide_turn, 0 )
		return true
	else
		VariableSet( g_key_highTide_turn, turn + 1 )
		return false
	end

end

function HighTide()
    MapObjectAction( 9, 1, MAP_ACTION_DONE)
    マップ全体の地形変化( "TID_砂浜_M016", "TID_浅瀬" )

    VariableSet(g_key_highTide,1);
    VariableSet(g_key_highTide_turn,-1);
end

function EbbTide()
    MapObjectAction( 9, 1, MAP_ACTION_IDLE)
    マップ全体の地形変化( "TID_浅瀬", "TID_砂浜_M016" )
    VariableSet(g_key_highTide,0);
    VariableSet(g_key_highTide_turn,4);
end

function HighTideByDragonVein()
     if not IsMauveAlive() then
        return
    end
    local x = UnitGetX(g_pid_Mauve)
    local z = UnitGetZ(g_pid_Mauve)
    CursorSetPos_FromPid( g_pid_Mauve)
    WaitTime( 0.5 )
    EffectPlay( "水球_発動", x, z )
    EffectWait()
    CursorSetPos( 15, 11 )
    MapCameraWait()

    EffectPlay( "水球", 15, 11 )
    EffectWait()
    HighTide()
end

function HighTideByTurn()
    Log("HighTideByTurn")
    if VariableGet(g_key_highTide) == 1 then
        return;
    end

    Log("HighTideByTurn2")
    if VariableGet(g_key_highTide) == 0 then
        local HighTideTurnCounter = VariableGet(g_key_highTide_turn);
        Log(HighTideTurnCounter)
        HighTideTurnCounter = HighTideTurnCounter - 1;
        VariableSet(g_key_highTide_turn,HighTideTurnCounter);
        Log(VariableGet(g_key_highTide_turn))
        if HighTideTurnCounter == 0 then
            HighTideByDragonVein()
        end
    end
end

function マップ全体の地形変化( fromTid, toTid )

	TerrainSetBegin()
	for x = 0, map_width-1 do
		for z = 0, map_height-1 do
			if TerrainGet(x, z) == fromTid then
				TerrainSet(x, z, toTid)
			end
		end
	end
	TerrainSetEnd()

end
