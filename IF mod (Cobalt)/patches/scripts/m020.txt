function RewrapReturn()

    local bossNowZ = UnitGetZ(g_pid_boss)

    if bossNowX ~= 10 or bossNowZ ~= 25 then
        CursorSetPos(bossNowX, bossNowZ)
        MapCameraWait()

        EffectPlay( "ワープアウト_闇", bossNowX, bossNowZ)
        EffectWait()

        CursorSetPos(10, 25)

end

function DragonVeinUsed()
    local unit = MindGetUnit()
    local x = UnitGetX(unit)
    local z = UnitGetZ(unit)

    local dragonVeinPoints = {}
    for _, config in ipairs(g_dragonVeinConfig_M020) do
        table.insert(dragonVeinPoints, {
            config.point,
            config.skillId,
            config.type,
            config.param
        })
    end

    local veinInfo = DragonVeinPreUseCheck(
        dragonVeinPoints,
        x,
        z
    )

    if veinInfo then
        DragonVeinPostUseHandleM020(veinInfo)
    end
end

function DragonVeinPostUseHandleM020(veinInfo)
    TerrainSetBegin()
        TerrainSet(veinInfo.x, veinInfo.z, "TID_床")
    TerrainSetEnd()

    EffectDelete("魔法陣_G001", veinInfo.x, veinInfo.z)
    CursorSetPos(veinInfo.x, veinInfo.z)
    WaitTime(1.0)
    EffectPlay("水球_発動", veinInfo.x, veinInfo.z)
    WaitTime(1.5)

    for _, config in ipairs(g_dragonVeinConfig_M020) do
        if veinInfo.x == config.point[1] and veinInfo.z == config.point[2] then
            local distOrg = CursorGetDistanceMode()

            CursorSetDistanceMode(CURSOR_DISTANCE_FAR)
            MapCameraWait()

            CursorSetPos((g_map_width+1)/2, (g_map_height+1)/2)
            WaitTime(1.0)
            EffectPlay("水球", (g_map_width+1)/2, (g_map_height+1)/2)
            WaitTime(1.5)
            AddDivineWindEffectToAllUnits()

            CursorSetDistanceMode(distOrg)
            MapCameraWait()

            MapRangeClear()
            VariableSet("RangeAttackStop_temp_triggered", 1)
            break
        end
    end

end

function AddDivineWindEffectToAllUnits()
    local playerIndex = ForceUnitGetFirst(FORCE_PLAYER)
    while playerIndex ~= nil do
        UnitSetPrivateSkill(playerIndex, "SID_神風招来効果")
        playerIndex = ForceUnitGetNext(playerIndex)
    end

    local enemyIndex = ForceUnitGetFirst(FORCE_ENEMY)
    while enemyIndex ~= nil do
        UnitSetPrivateSkill(enemyIndex, "SID_神風招来効果")
        enemyIndex = ForceUnitGetNext(enemyIndex)
    end
end

function RangeAddByTurn()
    MapRangeClear()
    WaitTime(1.0)

    if VariableGet("RangeAttackStop_triggered") == 1 then
        return
    end

    if UnitExistOnMap(g_pid_boss) == false then
        return
    end

    CursorSetPos(10,25)
    MapCameraWait()

    EffectPlay("邪竜紋_発動",10,25)
    WaitTime(1.0)

    local result =
        FindMostUnitsInSquareArea(g_map_width, g_map_height, 5, FORCE_PLAYER)
    attackRange = {}

    CursorSetPos(result.centerX,result.centerZ)
    MapCameraWait()

    local expandedSize = result.squareSize + 2
    local halfExpanded = (expandedSize - (expandedSize % 2)) / 2
    MapRangeAddBegin()
    for x = result.centerX - halfExpanded, result.centerX + halfExpanded do
        for z = result.centerZ - halfExpanded, result.centerZ + halfExpanded do
            if x >= 1 and x <= g_map_width and z >= 1 and z <= g_map_height and TerrainGetMoveCost(x, z) ~= "COST_不可"then
                MapRangeAdd(x, z)
                table.insert(attackRange,{targetX = x,targetZ = z})
            end
        end
    end
    MapRangeAddEnd()

    bossRewrapX = result.centerX
    bossRewrapZ = result.centerZ

    if TerrainGetMoveCost(bossRewrapX, bossRewrapZ) == "COST_不可" then
        local validX, validZ = findNearestValidPositionBFS(bossRewrapX, bossRewrapZ)
        if validX and validZ then
            bossRewrapX = validX
            bossRewrapZ = validZ
        end
    end

    if VariableGet("Tutorial_M020_Complete") == 0 then
        VariableSet("Tutorial_M020_Complete",1)
    end

    VariableSet("bossRewrapX", bossRewrapX)
    VariableSet("bossRewrapZ", bossRewrapZ)
end

function findNearestValidPositionBFS(centerX, centerZ)
    local queue = {{x = centerX, z = centerZ}}
    local visited = {}
    visited[centerX * 100 + centerZ] = true

    local directions = {{1,0}, {-1,0}, {0,1}, {0,-1}}

    while #queue > 0 do
        local current = table.remove(queue, 1)

        if current.x >= 1 and current.x <= g_map_width and
           current.z >= 1 and current.z <= g_map_height and
           TerrainGetMoveCost(current.x, current.z) ~= "COST_不可" then
            return current.x, current.z
        end

        for _, dir in ipairs(directions) do
            local newX = current.x + dir[1]
            local newZ = current.z + dir[2]
            local key = newX * 100 + newZ

            if newX >= 1 and newX <= g_map_width and
               newZ >= 1 and newZ <= g_map_height and
               not visited[key] then
                visited[key] = true
                table.insert(queue, {x = newX, z = newZ})
            end
        end
    end

    return nil, nil
end

function FindValidPosition(pid, targetX, targetZ, maxRange)
    local unit = UnitGetByPID(pid)
    if unit == nil then
        return nil, nil
    end

    for range = maxRange, 1, -1 do
        for dx = -range, range do
            for dz = -range, range do
                if math.abs(dx) + math.abs(dz) == range then
                    local x = targetX + dx
                    local z = targetZ + dz
                    if x >= 1 and x <= g_map_width and z >= 1 and z <= g_map_height then
                        local terrainCost = TerrainGetMoveCost(x, z)
                        if terrainCost ~= "COST_不可" then
                            local existingUnit = UnitGetByPos(x, z)
                            if existingUnit == nil then
                                return x, z
                            end
                        end
                    end
                end
            end
        end
    end

    return nil, nil
end



function RangeAttackByTurnInitValue()
    bossAttackList = {}

    for i, pos in ipairs(attackRange) do
        local targetX = pos.targetX
        local targetZ = pos.targetZ

        local unit = UnitGetByPos(targetX, targetZ)
        if unit ~= nil and UnitGetForce(unit) == FORCE_PLAYER then
            table.insert(bossAttackList,unit)
        end
    end


end

function BossAttackImpl()
    if VariableGet("RangeAttackStop_triggered") == 1 then
        return
    end

    if VariableGet("RangeAttackStop_temp_triggered") == 1 then
        UnitSetItemEquip("PID_IFMOD_M020_イルシオン兵_ハイプリースト_BOSS","IID_エルファイアー")
        UnitSetItemEquip("PID_IFMOD_M020_イルシオン兵_セイジ_BOSS","IID_エルファイアー")
        UnitSetItemEquip(g_pid_boss,"IID_エクスカリバー")
        VariableSet("RangeAttackStop_temp_triggered",0)
        return
    end

    if UnitExistOnMap(g_pid_boss) == false then
        return
    end

    local validTargets = {}
    bossRewrapXvar = VariableGet("bossRewrapX")
    bossRewrapZvar = VariableGet("bossRewrapZ")

    for i, targetUnit in ipairs(bossAttackList) do
        if targetUnit ~= nil and UnitIsExist(targetUnit) then
            local targetX = UnitGetX(targetUnit)
            local targetZ = UnitGetZ(targetUnit)

            local distanceX = math.abs(targetX - bossRewrapXvar)
            local distanceZ = math.abs(targetZ - bossRewrapZvar)

            if distanceX <= 3 and distanceZ <= 3 then
                table.insert(validTargets, targetUnit)
            end
        end
    end

    if #validTargets > 0 then
        CursorSetPos(UnitGetX(g_pid_boss), UnitGetZ(g_pid_boss))
        MapCameraWait()

        EffectPlay( "ワープアウト_闇", UnitGetX(g_pid_boss), UnitGetZ(g_pid_boss))
        EffectWait()

        CursorSetPos(bossRewrapXvar, bossRewrapZvar)
        MapCameraWait()

        EffectPlay( "ワープアウト_闇", bossRewrapXvar, bossRewrapZvar)
        EffectWait()

        local unit = UnitGetByPos(bossRewrapXvar,bossRewrapZvar)
        if unit ~= nil then
            UnitDie(unit)
        end

        UnitSetPos(g_pid_boss,bossRewrapXvar,bossRewrapZvar)
        AiSetRejectPower0Attack( g_pid_boss, false )
    end
end


function updateAttackList()
    for i = #bossInValidAttackList, 1, -1 do
        local unit = bossInValidAttackList[i]
        if unit ~= nil then
            local targetX = UnitGetX(unit)
            local targetZ = UnitGetZ(unit)

            local bossX, bossZ = FindValidPosition(g_pid_boss, targetX, targetZ, 2)

            if bossX ~= nil and bossZ ~= nil then
                table.remove(bossInValidAttackList, i)
                table.insert(bossAttackList, unit)
            end
        end
    end
end

function CheckReinforcementCondition_1()
	if VariableGet("reinforcement_1_triggered") == 1 then
		return false
	end

	local index = ForceUnitGetFirst(FORCE_PLAYER)
	while index ~= nil do
		local z = UnitGetZ(index)
		if z >= 11 then
			VariableSet("reinforcement_1_triggered", 1)
			return true
		end
		index = ForceUnitGetNext(index)
	end
	return false
end
