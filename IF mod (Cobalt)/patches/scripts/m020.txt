function RangeAddByTurn()
    local result =
        FindMostUnitsInSquareArea(g_map_width, g_map_height, 5, FORCE_PLAYER)--local index = ForceUnitGetFirst(FORCE_PLAYER)
    attackRange = {}
    local expandedSize = result.squareSize + 2
    local halfExpanded = (expandedSize - (expandedSize % 2)) / 2
    MapRangeAddBegin()
    for x = result.centerX - halfExpanded, result.centerX + halfExpanded do
        for z = result.centerZ - halfExpanded, result.centerZ + halfExpanded do
            if x >= 1 and x <= g_map_width and z >= 1 and z <= g_map_height and TerrainGetMoveCost(x, z) ~= "COST_不可"then
                MapRangeAdd(x, z)
                table.insert(attackRange,{targetX = x,targetZ = z})
            end
        end
    end
    MapRangeAddEnd()

    hasBeginAttack = 0
end

function FindValidPosition(pid, targetX, targetZ, maxRange)
    local unit = UnitGetByPID(pid)
    if unit == nil then
        return nil, nil
    end

    for range = maxRange, 1, -1 do
        for dx = -range, range do
            for dz = -range, range do
                if math.abs(dx) + math.abs(dz) == range then
                    local x = targetX + dx
                    local z = targetZ + dz
                    if x >= 1 and x <= g_map_width and z >= 1 and z <= g_map_height then
                        local terrainCost = TerrainGetMoveCost(x, z)
                        if terrainCost ~= "COST_不可" then
                            local existingUnit = UnitGetByPos(x, z)
                            if existingUnit == nil then
                                return x, z
                            end
                        end
                    end
                end
            end
        end
    end

    return nil, nil
end

function BossAttackImpl()
    if #bossAttackList == 0 and hasBeginAttack == 0 then
        return
    end

    if #bossAttackList == 0 and hasBeginAttack ~= 0 then
        return
    end

    hasBeginAttack = 1

    local index = ForceUnitGetFirst(FORCE_PLAYER)
    TerrainSetBegin()
        while index ~= nil do
            local indexX = UnitGetX(index)
            local indexZ = UnitGetZ(index)
            if TerrainGet(indexX,indexZ) ~= "TID_柱_Barrier" then
                TerrainSet(indexX,indexZ,"TID_柱_Barrier")
            end
            index = ForceUnitGetNext(index)
        end
    TerrainSetEnd()

    local targetUnit = table.remove(bossAttackList, 1)
    if targetUnit ~= nil then
        local targetX = UnitGetX(targetUnit)
        local targetZ = UnitGetZ(targetUnit)

        local bossX, bossZ = FindValidPosition(g_pid_boss, targetX, targetZ, 2)

        local bossStartX = UnitGetX(g_pid_boss)
        local bossStartZ = UnitGetZ(g_pid_boss)

        if bossX ~= nil and bossZ ~= nil then
            CursorSetPos(bossStartX,bossStartZ)
            MapCameraWait()

            EffectPlay("ワープアウト_闇",bossStartX,bossStartZ)
            MapCameraWait()

            CursorSetPos(bossX,bossZ)
            MapCameraWait()

            EffectPlay("ワープイン_闇",bossX,bossZ)
            UnitSetPos(g_pid_boss, bossX, bossZ)
            MapCameraWait()

            for i, terrainInfo in ipairs(unitTerrainMap) do
                Log(terrainInfo.Index)
                if terrainInfo.Index == targetUnit then
                    CobaltNotice(terrainInfo.Terrain)
                    Log("In set")
                    TerrainSetBegin()
                        TerrainSet(terrainInfo.targetX, terrainInfo.targetZ, terrainInfo.Terrain)
                    TerrainSetEnd()
                    break
                end
            end
        end
    end
end
