function initMapM020()
    local magicCirclePoints = {}
    for _, config in ipairs(g_dragonVeinConfig_M020) do
        table.insert(magicCirclePoints, config.point)
    end
    DragonVeinCreateMagicCircle(magicCirclePoints)

    local terrainPoints = {}
    for _, config in ipairs(g_dragonVeinConfig_M020) do
        if config.type == "silence" then
            table.insert(terrainPoints, {config.point[1], config.point[2], "TID_龙脉_M020_Silence"})
        end
    end
    DragonVeinSetTerrain(terrainPoints)
end

function EquipDragonVeinSkillsM020()

    for _, config in ipairs(g_dragonVeinConfig_M020) do
        table.insert(skillIds, config.skillId)
    end
    DragonVeinEquipSkills(skillIds)
end

function DragonVeinUsed()
    local unit = MindGetUnit()
    local x = UnitGetX(unit)
    local z = UnitGetZ(unit)

    local dragonVeinPoints = {}
    for _, config in ipairs(g_dragonVeinConfig_M020) do
        table.insert(dragonVeinPoints, {
            config.point,
            config.skillId,
            config.type,
            config.param
        })
    end

    local veinInfo = DragonVeinPreUseCheck(
        dragonVeinPoints,
        x,
        z
    )

    if veinInfo then
        DragonVeinPostUseHandleM020(veinInfo)
    end
end

function DragonVeinPostUseHandleM020(veinInfo)
    TerrainSetBegin()
        TerrainSet(veinInfo.x, veinInfo.z, "TID_床")
    TerrainSetEnd()

    EffectDelete("魔法陣_G001", veinInfo.x, veinInfo.z)
    CursorSetPos(veinInfo.x, veinInfo.z)
    WaitTime(1.0)
    EffectPlay("水球_発動", veinInfo.x, veinInfo.z)
    WaitTime(1.5)

    for _, config in ipairs(g_dragonVeinConfig_M020) do
        if veinInfo.x == config.point[1] and veinInfo.z == config.point[2] then
            local distOrg = CursorGetDistanceMode()

            CursorSetDistanceMode(CURSOR_DISTANCE_FAR)
            MapCameraWait()

            CursorSetPos((g_map_width+1)/2, (g_map_height+1)/2)
            WaitTime(1.0)
            EffectPlay("水球", (g_map_width+1)/2, (g_map_height+1)/2)
            AddDivineWindEffectToAllUnits()

            CursorSetDistanceMode(distOrg)
            MapCameraWait()

            MapRangeClear()
            VariableSet("RangeAttackStop_temp_triggered", 1)
            break
        end
    end

end

function AddDivineWindEffectToAllUnits()
    local playerIndex = ForceUnitGetFirst(FORCE_PLAYER)
    while playerIndex ~= nil do
        UnitSetPrivateSkill(playerIndex, "SID_神風招来効果")
        playerIndex = ForceUnitGetNext(playerIndex)
    end

    local enemyIndex = ForceUnitGetFirst(FORCE_ENEMY)
    while enemyIndex ~= nil do
        UnitSetPrivateSkill(enemyIndex, "SID_神風招来効果")
        enemyIndex = ForceUnitGetNext(enemyIndex)
    end
end

function RangeAddByTurn()
    MapRangeClear()
    WaitTime(1.0)

    if VariableGet("RangeAttackStop_triggered") == 1 then
        return
    end

    CursorSetPos(10,25)
    MapCameraWait()

    EffectPlay("邪竜紋_発動",10,25)
    WaitTime(1.0)

    local result =
        FindMostUnitsInSquareArea(g_map_width, g_map_height, 5, FORCE_PLAYER)--local index = ForceUnitGetFirst(FORCE_PLAYER)
    attackRange = {}

    CursorSetPos(result.centerX,result.centerZ)
    MapCameraWait()

    local expandedSize = result.squareSize + 2
    local halfExpanded = (expandedSize - (expandedSize % 2)) / 2
    MapRangeAddBegin()
    for x = result.centerX - halfExpanded, result.centerX + halfExpanded do
        for z = result.centerZ - halfExpanded, result.centerZ + halfExpanded do
            if x >= 1 and x <= g_map_width and z >= 1 and z <= g_map_height and TerrainGetMoveCost(x, z) ~= "COST_不可"then
                MapRangeAdd(x, z)
                table.insert(attackRange,{targetX = x,targetZ = z})
            end
        end
    end
    MapRangeAddEnd()

    bossRewrapX = result.centerX
    bossRewrapZ = result.centerZ

    if VariableGet("Tutorial_M020_Complete") == 0 then
        VariableSet("Tutorial_M020_Complete",1)
    end
end

function FindValidPosition(pid, targetX, targetZ, maxRange)
    local unit = UnitGetByPID(pid)
    if unit == nil then
        return nil, nil
    end

    for range = maxRange, 1, -1 do
        for dx = -range, range do
            for dz = -range, range do
                if math.abs(dx) + math.abs(dz) == range then
                    local x = targetX + dx
                    local z = targetZ + dz
                    if x >= 1 and x <= g_map_width and z >= 1 and z <= g_map_height then
                        local terrainCost = TerrainGetMoveCost(x, z)
                        if terrainCost ~= "COST_不可" then
                            local existingUnit = UnitGetByPos(x, z)
                            if existingUnit == nil then
                                return x, z
                            end
                        end
                    end
                end
            end
        end
    end

    return nil, nil
end



function RangeAttackByTurnInitValue()
    bossAttackList = {}

    for i, pos in ipairs(attackRange) do
        local targetX = pos.targetX
        local targetZ = pos.targetZ

        local unit = UnitGetByPos(targetX, targetZ)
        if unit ~= nil and UnitGetForce(unit) == FORCE_PLAYER then
            table.insert(bossAttackList,unit)
        end
    end


end

function updateAttackList()
    for i = #bossInValidAttackList, 1, -1 do
        local unit = bossInValidAttackList[i]
        if unit ~= nil then
            local targetX = UnitGetX(unit)
            local targetZ = UnitGetZ(unit)

            local bossX, bossZ = FindValidPosition(g_pid_boss, targetX, targetZ, 2)

            if bossX ~= nil and bossZ ~= nil then
                table.remove(bossInValidAttackList, i)
                table.insert(bossAttackList, unit)
            end
        end
    end
end
