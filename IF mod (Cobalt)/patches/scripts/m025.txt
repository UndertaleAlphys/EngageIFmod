function InitMapM025()
    Log("InitMapM025")
    for _ , effect in ipairs(CreateEffectsList) do
        EffectCreateRotateXYZ("魔法障壁_展開_E005", effect[1], effect[2], effect[3], effect[4], effect[5], effect[6], effect[7])
    end
	EventActionObject(8, 17, MAP_ACTION_DONE)
	EventActionObject(18, 17, MAP_ACTION_DONE)
end

function LightningDestroy(areaIndex)
    local effects = lightningEffectsMap[areaIndex]
    if effects then
        for _, coord in ipairs(effects) do
            EffectDelete("魔法障壁_展開_E005", coord[1], coord[2])
        end
    end

	for _, mapping in ipairs(areaIndexReinforcementMap) do
        if mapping[1] == areaIndex then
            local reinforcementKey = "g_reinforcement_" .. mapping[2]
			Log(reinforcementKey)
            VariableSet(reinforcementKey, 1)
            break
        end
    end

	if (areaIndex == 7 or areaIndex == 8) and VariableGet("g_reinforcement_m1_multiflag") == 0 then
        VariableSet("g_reinforcement_m1", 1)
		VariableSet("g_reinforcement_m1_multiflag", 1)
        Log("g_reinforcement_m1")
    end
    
	if areaIndex == 1 then
        VariableSet(g_key_band1_active, 1)
	elseif areaIndex == 2 then
        VariableSet(g_key_band2_active, 1)
	elseif areaIndex == 3 then
        VariableSet(g_key_band3_active, 1)
	elseif areaIndex == 4 then
        VariableSet(g_key_band4_active, 1)
	elseif areaIndex == 5 then
        VariableSet(g_key_band5_active, 1)
	elseif areaIndex == 6 then
        VariableSet(g_key_band6_active, 1)
	elseif areaIndex == 7 then
        VariableSet(g_key_band7_active, 1)
	elseif areaIndex == 8 then
        VariableSet(g_key_band7_active, 1)
	elseif areaIndex == 9 then
        VariableSet(g_key_band9_active, 1)
	elseif areaIndex == 10 then
        VariableSet(g_key_band9_active, 1)
	elseif areaIndex == 11 then
        VariableSet(g_key_band9_active, 1)
	end
end

function GenericReinforcement(reinforcementVar, disposName, logName)
    if VariableGet(reinforcementVar) == 0 then
        return
    end
    
    Dispos(disposName, DISPOS_FLAG_FOCUS)
    Yield()
    WaitTime(0.5)
    
    VariableSet(reinforcementVar, 0)
    Log(logName)
end

function SetAiRod1()
	if UnitExistOnMap("PID_M025_ルミエル") == false then
		SetAiRod2()
		return
	end

	if IsExistForcePlayerNearby("PID_M025_ルミエル",10) == false then
		UnitSetStatus( "PID_M025_ルミエル", UNIT_STATUS_FIXED )
	end

	if MapGetTurn()%4 == 1 then
		Set_Interference_AI_by_Pid("PID_M025_ルミエル")
	else
		AiSetSequence( "PID_M025_ルミエル", AI_ORDER_ATTACK, "AI_AT_Interference", "" )
	end

	if IsExistForcePlayerNearby("PID_M025_ルミエル",2) == true then
		AiSetSequence( "PID_M025_ルミエル", AI_ORDER_ATTACK, "AI_AT_Attack", "" )
	end
end

function 離脱イベント()
	local unit = MindGetUnit()
	if not UnitExistOnMap( unit ) then
		return
	end
	
	UnitRotation( g_pid_lueur, ROTATE_UP)
	UnitMoveWait()
	
	VariableSet( "勝利", 1 )
end

function OccupyVictory()
	local unit = MindGetUnit()
	if not UnitExistOnMap( unit ) then
		return
	end
	
	UnitMoveWait()
	
	VariableSet( "勝利", 1 )
end
