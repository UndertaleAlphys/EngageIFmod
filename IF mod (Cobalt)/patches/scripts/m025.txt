function ReEquipBossRod()
	local turn = MapGetTurn()
	local turnType = (turn - 1) % 4
	if turnType == 0 then
		turnType = 4
	end

	for index, rod in ipairs(bossRodList) do
		if turnType != index then
			ItemGainSilent( "PID_M025_ルミエル", rod )
		end
	end
end

function PutOffBossRod()
	Log("PutOffBossRod")
	local turn = MapGetTurn()
	local turnType = turn % 4
	if turnType == 0 then
		turnType = 4
	end

	for index, rod in ipairs(bossRodList) do
		if turnType != index then
			Log(index)
			UnitPutOffItem("PID_M025_ルミエル", rod)
		end
	end
end

function 離脱イベント()
	local unit = MindGetUnit()
	if not UnitExistOnMap( unit ) then
		return
	end
	
	UnitRotation( g_pid_lueur, ROTATE_UP)
	UnitMoveWait()
	
	VariableSet( "勝利", 1 )
end
