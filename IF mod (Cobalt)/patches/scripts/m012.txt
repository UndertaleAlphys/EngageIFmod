function quickSandTerrainSet()
	TerrainSetBegin()
	TerrainSet( 1, 8, "TID_王座")
	for x = 1, g_Width do
		for z = 1, g_Height do
			local direction = getMoveDirection()
			if TerrainGet(x, z) == "TID_流砂" or TerrainGet(x, z) ==  "TID_流砂_L" or TerrainGet(x, z) ==  "TID_流砂_R" or TerrainGet(x, z) ==  "TID_流砂_U" or TerrainGet(x, z) ==  "TID_流砂_D" then 
				if direction == "L" or direction == "左" then
					TerrainSet( x, z, "TID_流砂_D")
				elseif direction == "R" or direction == "右" then
					TerrainSet( x, z, "TID_流砂_U")
				elseif direction == "U" or direction == "上" then
					TerrainSet( x, z, "TID_流砂_L")
				elseif direction == "D" or direction == "下" then
					TerrainSet( x, z, "TID_流砂_R")
				end
			end
		end
	end
	TerrainSetEnd()
end

function 飛行ユニットの０ダメージ攻撃を許可()
	if DifficultyGet() == DIFFICULTY_LUNATIC then
		
		local index	= ForceUnitGetFirst( FORCE_ENEMY )
		
		while ( index != nil ) do
			
			if (UnitGetPID( index ) == "PID_M012_IFMOD_ランスファイター" ) then
				AiSetRejectPower0Attack( index, false )
			end
			
			index = ForceUnitGetNext( index )
		end
	end
end

function UnitMove()
    local direction = getMoveDirection()
    Log("Direction: " .. direction)

    local table_unit = moveListGet()
    if #table_unit == 0 then
        return
    end

    -- カメラを中央に設定
    local center_x, center_z = 中心点を算出(table_unit)
    CursorSetPos(center_x, center_z)
    MapCameraWait()

    -- 方向によって移動ベクトルを設定
    local dx, dz = 0, 0
    if direction == "L" or direction == "左" then
        dx = -1
    elseif direction == "R" or direction == "右" then
        dx = 1
    elseif direction == "U" or direction == "上" then
        dz = 1
    elseif direction == "D" or direction == "下" then
        dz = -1
    else
        Log("Invalid direction")
        return
    end

    -- 各ユニットを処理
    for i = 1, #table_unit do
        local t = table_unit[i]
        local x, z = t.X, t.Z
        local moved = false
	
        for step = 1, 5 do
            local nextX = x + dx 
            local nextZ = z + dz 


            -- マップ外チェック
            if nextX < 1 or nextX >= g_Width or nextZ < 1 or nextZ >= g_Height then
                break
            end

            -- 地形コストのチェック
            local cost = TerrainGetMoveCost(nextX, nextZ)
	    if isFlyCombatType(table_unit[i].UNIT) == false then
            	if cost == "COST_空" or cost == "COST_不可" then
                	break
            	end
	   end

	   if isFlyCombatType(table_unit[i].UNIT) == true then
		Log("Fly!!!")
            	if cost == "COST_不可" then
                	break
            	end
	   end

            -- 単位がすでにいるか？
            local unit = UnitGetByPos(nextX, nextZ)
            if unit ~= nil then
                break
            end

            -- 移動先がブロックかどうか
            local overlap = MapOverlapGet(nextX, nextZ)
            if overlap == "TID_ブロック" then
                break
            end

            moved = true
            x, z = nextX, nextZ
        end

        -- 実際に移動
        if moved then
            UnitMovePos(t.UNIT, x, z, MOVE_FLAG_NONE)
        end
    end

    -- 全員の移動完了まで待つ
    UnitMoveWait()

    -- 方向を移動方向に合わせる
    local rotation = ROTATE_RIGHT
    if direction == "L" or direction == "左" then
        rotation = ROTATE_LEFT
    elseif direction == "U" or direction == "上" then
        rotation = ROTATE_UP
    elseif direction == "D" or direction == "下" then
        rotation = ROTATE_DOWN
    end

    for i = 1, #table_unit do
        UnitRotation(table_unit[i].UNIT, rotation)
    end

    UnitMoveWait()
    MapCameraWait()
end

-- ------------------------------------
function getMoveDirection()
    local turn = MapGetTurn()
    local remainder = turn % 4

    local direction = ""

    if remainder == 1 then
        direction = "L"
    elseif remainder == 2 then
        direction = "D"
    elseif remainder == 3 then
        direction = "R"
    else -- remainder == 0
        direction = "U"
    end

    return direction
end

-- ------------------------------------
function moveListGet()
	local table_unit = {}
	local direction = getMoveDirection()

	local index = ForceUnitGetFirst(FORCE_PLAYER)
	while index != nil do
		
		local _x = UnitGetX(index)
		local _z = UnitGetZ(index)	

		local terrain = TerrainGet(_x, _z)
		if (terrain == "TID_流砂_L") or (terrain == "TID_流砂_R") or (terrain == "TID_流砂_U") or (terrain == "TID_流砂_D") then
			table_unit[ #table_unit + 1 ] = { UNIT=index, X=_x, Z=_z }
		end

		index = ForceUnitGetNext(index)
	
	end

	 if direction == "左" or direction == "L" then
        	table.sort(table_unit, function(a, b) return a.X < b.X end)
   	 elseif direction == "右" or direction == "R" then
        	table.sort(table_unit, function(a, b) return a.X > b.X end)
    	elseif direction == "上" or direction == "U" then
        	table.sort(table_unit, function(a, b) return a.Z > b.Z end)
    	elseif direction == "下" or direction == "D" then
        	table.sort(table_unit, function(a, b) return a.Z < b.Z end)
    	end
	
	return table_unit
end

-- ---------------------------------------------------------------
function 中心点を算出( table_unit )
	
	local x = 0
	local z = 0
	local count = 0
	local center_x = nil
	local center_z = nil
	
	if ( #table_unit > 0 ) then
			for i=1, #table_unit do
					local t = table_unit[i]
					x = ( x + t.X )
					z = ( z + t.Z )
					count = count + 1
			end
			
			center_x = math.floor(x/count + 0.5)
			center_z = math.floor(z/count + 0.5)
	end
	
	return center_x, center_z
	
end

function isFlyCombatType(pid)
	if ( UnitGetMoveCost( pid ) == "COST_飛行" ) then
		return true
	else
		return false
	end
end
