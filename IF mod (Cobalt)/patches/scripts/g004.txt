function condition_final_reinforcement()
    if VariableGet(has_reinforcement_final) == 1 then
        return false
    end

    if ForceUnitGetCount( FORCE_ENEMY ) <= 7 or MapGetTurn() >= 14 then
        VariableSet(has_reinforcement_final , 1)
        return true
    end
    return false
end

function volcanicBomb(x,z,size,map_damage)
    CursorSetPos( x, z )
    CursorSetDistanceMode( CURSOR_DISTANCE_NEAR )
    MapCameraWait()

    EffectPlay( "火山弾_G004", x, z )

    local expand = (size - 1)/2
    Log(expand)

    for i = x-expand,x+expand do
        for j = z-expand,z+expand do
            local unit = UnitGetByPos(i,j)
             if unit != nil then
                MapDamageBegin()
                local currentHp = UnitGetHp(unit)
                local damage = map_damage
                if damage >= currentHp then
                    damage = currentHp - 1
                end
                MapDamageAdd(unit, damage)
                MapDamageEnd()
            end
        end
    end

    EffectWait()
end

function volcanicBombWithSectionAnimations(x, z, size, map_damage)

    -- Determine which section the coordinates belong to
    local section = getMapSection(x, z)

    -- Play animation from the corresponding volcano based on section
    local volcanoPos = g_rockInSpot[section]
    CursorSetPos(volcanoPos[1], volcanoPos[2])

    local distOrg = CursorGetDistanceMode()

    CursorSetDistanceMode(CURSOR_DISTANCE_NEAR)
    MapCameraWait()
    if volcanoPos then
        EffectPlay("火山弾_噴出_G004", volcanoPos[1], volcanoPos[2])
    else
        -- Fallback to center position if section not found
        EffectPlay("火山弾_噴出_G004", x, z)
    end

    EffectWait()

    CursorSetPos( x, z )
    CursorSetDistanceMode( CURSOR_DISTANCE_FAR )
    MapCameraWait()

    EffectPlay( "火山弾_G004", x, z )

    local expand = (size - 1) / 2

    WaitTime(0.5)
    EffectWait()

    local damagedPositions = {}
    for i = x - expand, x + expand do
        for j = z - expand, z + expand do
            local unit = UnitGetByPos(i, j)
            if unit ~= nil and UnitGetForce(unit) == FORCE_PLAYER then
                MapDamageBegin()
                local currentHp = UnitGetHp(unit)
                local damage = map_damage
                if damage >= currentHp then
                    damage = currentHp - 1
                end
                MapDamageAdd(unit, damage)
                MapDamageEnd()
            end
            table.insert(damagedPositions, {x = i, z = j})
        end
    end

    if #damagedPositions > 0 then
        MapOverlapSetBegin()
        for _, pos in ipairs(damagedPositions) do
            MapOverlapSet(pos.x, pos.z, "TID_炎上")
        end
        MapOverlapSetEnd()
    end

    CursorSetDistanceMode(distOrg)
end

function volcanicBombWithSectionAnimationsDragonVein(x, z, size, map_damage)

    local distOrg = CursorGetDistanceMode()

    CursorSetDistanceMode(CURSOR_DISTANCE_NEAR)
    MapCameraWait()
    CursorSetPos( x, z )
    MapCameraWait()

    EffectPlay( "火山弾_G004", x, z )
    EffectWait()
    local expand = (size - 1) / 2

    local damagedPositions = {}
    for i = x - expand, x + expand do
        for j = z - expand, z + expand do
            local unit = UnitGetByPos(i, j)
            if unit ~= nil and UnitGetForce(unit) == FORCE_ENEMY then
                MapDamageBegin()
                local currentHp = UnitGetHp(unit)
                local damage = map_damage
                if damage >= currentHp then
                    damage = currentHp - 1
                end
                MapDamageAdd(unit, damage)
                MapDamageEnd()
            end
            table.insert(damagedPositions, {x = i, z = j})
        end
    end

    if #damagedPositions > 0 then
        MapOverlapSetBegin()
        for _, pos in ipairs(damagedPositions) do
            MapOverlapSet(pos.x, pos.z, "TID_炎上")
        end
        MapOverlapSetEnd()
    end

    CursorSetDistanceMode(distOrg)
end

-- Determine which section of the map the coordinates belong to
function getMapSection(x, z)
    local sectionX = 0
    local sectionZ = 0

    -- Calculate section boundaries (assuming 4 equal sections on a 25x32 map)
    local halfWidth = g_map_width / 2
    local halfHeight = g_map_height / 2

    -- Determine horizontal section (1=left, 2=right)
    if x <= halfWidth then
        sectionX = 1
    else
        sectionX = 2
    end

    -- Determine vertical section (1=top, 2=bottom)
    if z > halfHeight then
        sectionZ = 1
    else
        sectionZ = 2
    end

    -- Map to our AREA constants:
    -- AREA_A (1) = Top-left
    -- AREA_B (2) = Top-right
    -- AREA_C (3) = Bottom-left
    -- AREA_D (4) = Bottom-right
    if sectionX == 1 and sectionZ == 1 then
        return AREA_A
    elseif sectionX == 2 and sectionZ == 1 then
        return AREA_B
    elseif sectionX == 1 and sectionZ == 2 then
        return AREA_C
    elseif sectionX == 2 and sectionZ == 2 then
        return AREA_D
    end

    -- Default to AREA_A if something goes wrong
    return AREA_A
end

function getRangeSquareWithMapDamage(squareSize, map_height, map_width, map_damage)

    local damageGrid = {}
    for z = 1, map_height - squareSize + 1 do
        damageGrid[z] = {}
        for x = 1, map_width - squareSize + 1 do
            damageGrid[z][x] = 0
        end
    end

    for z = 1, map_height do
        for x = 1, map_width do
            local unit = UnitGetByPos(x, z)
            if unit ~= nil and UnitGetForce(unit) == FORCE_PLAYER then
                local currentHp = UnitGetHp(unit)
                local damage = map_damage

                if damage >= currentHp then
                    damage = currentHp - 1
                end

                local pid = UnitGetPID(unit)

                local minSquareZ = math.max(1, z - squareSize + 1)
                local maxSquareZ = math.min(map_height - squareSize + 1, z)
                local minSquareX = math.max(1, x - squareSize + 1)
                local maxSquareX = math.min(map_width - squareSize + 1, x)

                for sz = minSquareZ, maxSquareZ do
                    for sx = minSquareX, maxSquareX do
                        damageGrid[sz][sx] = damageGrid[sz][sx] + damage + 10
                    end
                end
            end
        end
    end

    local maxDamageResult = {}
    local maxTotalDamage = 0

    for z = 1, map_height - squareSize + 1 do
        for x = 1, map_width - squareSize + 1 do
            local totalDamage = damageGrid[z][x]
            if totalDamage > maxTotalDamage then
                maxTotalDamage = totalDamage
                local centerX = x + (squareSize - 1) * 0.5
                local centerZ = z + (squareSize - 1) * 0.5
                maxDamageResult = {
                    x = x,
                    z = z,
                    centerX = centerX,
                    centerZ = centerZ,
                    playerUnitCount = 0,
                    totalDamage = totalDamage
                }
            end
        end
    end

    return maxDamageResult
end
